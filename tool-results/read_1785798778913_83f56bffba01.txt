    76→// Local helper function to format dates for input fields (YYYY-MM-DD)
    77→const formatDateLocal = (dateString?: string) => {
    78→  if (!dateString) return ""
    79→  const date = parseDateWithTimezone(dateString)
    80→  const year = date.getFullYear()
    81→  const month = String(date.getMonth() + 1).padStart(2, '0')
    82→  const day = String(date.getDate()).padStart(2, '0')
    83→  return `${year}-${month}-${day}`
    84→}
    85→
    86→// Local helper function to format dates to UTC midnight
    87→const formatDateToUTC = (dateString: string | null | undefined) => {
    88→  if (!dateString) return null;
    89→  const date = new Date(dateString);
    90→  // Set to UTC midnight
    91→  return new Date(Date.UTC(date.getFullYear(), date.getMonth(), date.getDate())).toISOString();
    92→};
