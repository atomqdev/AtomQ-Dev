    49→const formatDateDDMMYYYY = (dateString: string) => {
    50→  const date = new Date(dateString)
    51→  const day = String(date.getDate()).padStart(2, '0')
    52→  const month = String(date.getMonth() + 1).padStart(2, '0')
    53→  const year = date.getFullYear()
    54→  return `${day}/${month}/${year}`
    55→}

   264→        method: "PATCH",

   267→        },
   268→        body: JSON.stringify({
   269→          reportId,

   298→      reference: report.question.reference,
   299→      title: report.question.title,
   300→      type: report.question.type,
   301→      options: parsedOptions,
   302→      correctAnswer: report.question.correctAnswer,
   303→      correctAnswers,
   304→      explanation: report.question.explanation || "",
   305→      difficulty: report.question.difficulty,
   306→      isActive: report.question.isActive

   309→  }
   310→
   311→  const handleSubmit = async (e: React.FormEvent) => {
   312→    e.preventDefault()
   313→
   314→    if (!formData.reference.trim()) {
   315→      toast.error("Reference is required")
   316→      return
   317→    }
   318→
   319→    const tempDiv = document.createElement('div')
   320→    tempDiv.innerHTML = formData.title
   321→    const textContent = tempDiv.textContent || tempDiv.innerText || ""
   322→
   323→    if (!textContent.trim()) {
   324→      toast.error("Title is required")
   325→      return
   326→    }
   327→
   328→    if (formData.type !== QuestionType.FILL_IN_BLANK) {
   329→      if (formData.options.length === 0) {
   330→        toast.error("At least one option is required")
   331→        return

   333→
   334→      if (formData.options.some(option => !option.trim())) {
   335→        toast.error("All options must have values")
   336→        return
   337→      }
   338→
   339→      if (formData.type === QuestionType.MULTI_SELECT) {
   340→        if (formData.correctAnswers.length === 0) {
   341→          toast.error("At least one correct answer must be selected for multi-select questions")
   342→          return
   343→        }
   344→      } else {
   345→        if (!formData.correctAnswer.trim()) {
   346→          toast.error("A correct answer must be selected")
   347→          return
   348→        }
   349→      }
   350→    } else {
   351→      if (!formData.correctAnswer.trim()) {
   352→        toast.error("Correct answer is required for fill-in-the-blank questions")
   353→        return
   354→      }
   355→    }
   356→
   357→    setSubmitLoading(true)
   358→
   359→    try {
   360→      const apiData = {
   361→        reference: formData.reference,
   362→        title: formData.title,
   363→        type: formData.type,
   364→        options: formData.options,
   365→        correctAnswer: formData.correctAnswer,
   366→        explanation: formData.explanation,
   367→        difficulty: formData.difficulty,
   368→        isActive: formData.isActive
   369→      }
   370→
   371→      const url = `/api/admin/question-groups/${params.id}/questions/${editingQuestion.id}`
   372→
   373→      const response = await fetch(url, {
   374→        method: "PUT",
   375→        headers: {
   376→          "Content-Type": "application/json"
   377→        },
   378→        body: JSON.stringify(apiData)
   379→      })
   380→
   381→      if (response.ok) {
   382→        await fetchReportedQuestions()
   383→        setIsEditDialogOpen(false)
   384→        resetForm()
   385→        toast.success("Question updated successfully")
   386→      } else {
   387→        const errorData = await response.json()
   388→        toast.error(`Error: ${errorData.message}`)
   389→      }
   390→    } catch (error) {
   391→      console.error("Error saving question:", error)
   392→      toast.error("Error saving question. Please try again.")
   393→    } finally {
   394→      setSubmitLoading(false)
   395→    }
   396→  }
   397→
   398→  const resetForm = () => {
   399→    setEditingQuestion(null)
   400→    setFormData({
   401→      reference: "",
   402→      title: "",
   403→      type: QuestionType.MULTIPLE_CHOICE as QuestionType,
   404→      options: ["", "", ""],
   405→      correctAnswer: "",
   406→      correctAnswers: [],
   407→      explanation: "",
   408→      difficulty: DifficultyLevel.MEDIUM as DifficultyLevel,
   409→      isActive: true

   411→  }

   449→        ...formData,
   450→        correctAnswers: newCorrectAnswers,
   451→        correctAnswer: newCorrectAnswers.join('|')

   454→      setFormData({
   455→        ...formData,

   469→        return "Fill in Blank"

   472→      default:
   473→        return type
   474→    }
   475→  }

   484→        <div>
   485→          <h1 className="text-3xl font-bold tracking-tight">Reported Questions</h1>
   486→          <p className="text-muted-foreground">
   487→            Review and manage reported questions for this question group
   488→          </p>
   489→        </div>
   490→        <Button variant="outline" onClick={() => router.back()}>
   491→          <ArrowLeft className="h-4 w-4" />

   494→
   495→      <Card>
   496→        <CardContent>

   498→            columns={columns}

   526→          {selectedReport && (
   527→            <div className="space-y-6">
   528→              <div className="grid grid-cols-2 gap-4">
   529→                <div>
   530→                  <h4 className="font-semibold text-sm text-muted-foreground mb-1">Reported By</h4>
   531→                  <p className="font-medium">{selectedReport.user.name || "Unknown"}</p>
   532→                  <p className="text-sm text-muted-foreground">{selectedReport.user.email}</p>
   533→                </div>
   534→                <div>
   535→                  <h4 className="font-semibold text-sm text-muted-foreground mb-1">Reported At</h4>
   536→                  <p className="font-medium">
   537→                    {format(new Date(selectedReport.createdAt), "MMM d, yyyy h:mm a")}
   538→                  </p>
   539→                </div>
   540→              </div>
   541→
   542→              <div>
   543→                <h4 className="font-semibold text-sm text-muted-foreground mb-2">User Suggestion</h4>
   544→                <div className="bg-muted/50 p-4 rounded-lg">
   545→                  <RichTextDisplay content={selectedReport.suggestion} />
   546→                </div>
   547→              </div>
   548→
   549→              <div>
   550→                <h4 className="font-semibold text-sm text-muted-foreground mb-2">Question</h4>
   551→                <div className="bg-muted/50 p-4 rounded-lg space-y-3">

   553→                    <h5 className="font-medium">{selectedReport.question.reference}</h5>
   554→                    <RichTextDisplay content={selectedReport.question.title} />
   555→                  </div>
   556→                  <div className="grid grid-cols-3 gap-4 text-sm">

   561→                    <div>
   562→                      <span className="font-medium">Difficulty:</span>
   563→                      <span className="ml-2">{selectedReport.question.difficulty}</span>
   564→                    </div>
   565→                    <div>
   566→                      <span className="font-medium">Correct Answer:</span>
   567→                      <span className="ml-2">{selectedReport.question.correctAnswer}</span>
   568→                    </div>
   569→                  </div>
   570→                </div>
   571→              </div>
   572→
   573→              {selectedReport.question.explanation && (
   574→                <div>
   575→                  <h4 className="font-semibold text-sm text-muted-foreground mb-2">Explanation</h4>

   589→                    {updating === selectedReport.id ? (

   591→                        <Loader2 className="mr-2 h-4 w-4 animate-spin" />

   597→                  </Button>
   598→                  <Button
   599→                    onClick={() => handleMarkAsResolved(selectedReport.id)}
   600→                    disabled={updating === selectedReport.id}
   601→                  >
   602→                    {updating === selectedReport.id ? (
   603→                      <>
   604→                        <Loader2 className="mr-2 h-4 w-4 animate-spin" />
   605→                        Marking Resolved...
   606→                      </>
   607→                    ) : (
   608→                      "Mark as Resolved"
   609→                    )}
