   136→function SortableQuestion({
   137→  question,
   138→  onEdit,
   139→  onDelete,
   140→  onView
   141→}: {
   142→  question: Question
   143→  onEdit: (question: Question) => void
   144→  onDelete: (questionId: string) => void
   145→  onView: (question: Question) => void
   146→}) {
   147→  const {
   148→    attributes,
   149→    listeners,
   150→    setNodeRef,
   151→    transform,
   152→    transition,
   153→    isDragging,
   154→  } = useSortable({ id: question.id })
   155→
   156→  const style = {
   157→    transform: CSS.Transform.toString(transform),
   158→    transition,
   159→    opacity: isDragging ? 0.5 : 1,
   160→  }
   161→
   162→  return (
   163→    <TableRow ref={setNodeRef} style={style} {...attributes}>
   164→      <TableCell>
   165→        <div {...listeners} className="cursor-grab">
   166→          <GripVertical className="h-4 w-4" />
   167→        </div>
   168→      </TableCell>
   169→      <TableCell className="font-medium">{question.order}</TableCell>
   170→      <TableCell>
   171→        <div>
   172→          <div className="font-medium">{question.title}</div>
   173→          {/* <div className="text-sm text-muted-foreground">
   174→            <RichTextDisplay content={question.content} />
   175→          </div> */}
   176→        </div>
   177→      </TableCell>
   178→      <TableCell>
   179→        <Badge variant={
   180→          question.type === QuestionType.MULTIPLE_CHOICE ? "default" :
   181→            question.type === QuestionType.MULTI_SELECT ? "secondary" :
   182→            question.type === QuestionType.TRUE_FALSE ? "outline" : "destructive"
   183→        }>
   184→          {question.type.replace('_', ' ')}
   185→        </Badge>
   186→      </TableCell>
   187→      <TableCell>
   188→        <Badge variant={
   189→          question.difficulty === DifficultyLevel.EASY ? "default" :
   190→            question.difficulty === DifficultyLevel.MEDIUM ? "secondary" : "destructive"
   191→        }>
   192→          {question.difficulty}
   193→        </Badge>
   194→      </TableCell>
   195→      <TableCell>{question.points}</TableCell>
   196→      <TableCell>
   197→        <div className="flex gap-2">
   198→          <Button
   199→            variant="ghost"
   200→            size="sm"
   201→            onClick={() => onView(question)}
   202→          >
   203→            <Eye className="h-4 w-4" />
   204→          </Button>
   205→
   206→          <AlertDialog>
   207→            <AlertDialogTrigger asChild>
   208→              <Button
   209→                variant="ghost"
   210→                size="sm"
   211→                className="text-red-600"
   212→              >
   213→                <Trash2 className="h-4 w-4" />
   214→              </Button>
   215→            </AlertDialogTrigger>
   216→            <AlertDialogContent>
   217→              <AlertDialogHeader>
   218→                <AlertDialogTitle>Remove Question</AlertDialogTitle>
   219→                <AlertDialogDescription>
   220→                  Are you sure you want to remove this question from the assessment? This action cannot be undone.
   221→                </AlertDialogDescription>
   222→              </AlertDialogHeader>
   223→              <AlertDialogFooter>
   224→                <AlertDialogCancel>Cancel</AlertDialogCancel>
   225→                <AlertDialogAction onClick={() => onDelete(question.id)}>
   226→                  Remove
   227→                </AlertDialogAction>
   228→              </AlertDialogFooter>
   229→            </AlertDialogContent>
   230→          </AlertDialog>
   231→        </div>
   232→      </TableCell>
   233→    </TableRow>
   234→  )
   235→}

   433→  }
   434→
   435→  const handleExportQuestions = () => {
   436→    const csvContent = [
   437→      ["Title", "Content", "Type", "Options", "Correct Answer", "Explanation", "Difficulty", "Points"],
   438→      ...questions.map(question => {
   439→        // Parse options from database (stored as JSON string) and convert to pipe-separated format

   442→          const parsedOptions = Array.isArray(question.options)
   443→            ? question.options
   444→            : JSON.parse(question.options || "[]")
   445→          optionsString = parsedOptions.join("|")

   451→        return [
   452→          question.title,
   453→          question.content,
   454→          question.type,
   455→          optionsString,
   456→          question.correctAnswer,
   457→          question.explanation || "",
   458→          question.difficulty,
   459→          question.points.toString()

   528→  const handleImportWithGroup = async () => {

   541→    Papa.parse(importFile, {

   544→      complete: async (results) => {
   545→        try {
   546→
   547→          // Filter out empty rows and validate required fields
   548→          const validQuestions = results.data.filter((row: any) => {
   549→            const hasTitle = row.Title && row.Title.trim() !== ""
   550→            const hasContent = row.Content && row.Content.trim() !== ""
   551→            const hasType = row.Type && row.Type.trim() !== ""
   552→            const hasOptions = row.Options && row.Options.trim() !== ""
   553→            const hasCorrectAnswer = row["Correct Answer"] && row["Correct Answer"].trim() !== ""
   554→
   555→            return hasTitle && hasContent && hasType && hasOptions && hasCorrectAnswer
   556→          })
   557→
   558→
   559→          if (validQuestions.length === 0) {
   560→            toast.error("No valid questions found in CSV file. Please ensure all required fields are filled.")
   561→            return
   562→          }
   563→
   564→          // Create questions in the selected group
   565→          const importPromises = validQuestions.map(async (question: any, index: number) => {
   566→            try {
   567→
   568→              // Normalize question type
   569→              let questionType = question.Type?.toString().toUpperCase().trim()
   570→              if (!Object.values(QuestionType).includes(questionType as QuestionType)) {
   571→                questionType = QuestionType.MULTIPLE_CHOICE
   572→                console.warn(`Invalid question type "${question.Type}", defaulting to MULTIPLE_CHOICE`)
   573→              }
   574→
   575→              // Parse options
   576→              let options = []
   577→              if (question.Options) {
   578→                const optionsStr = question.Options.toString().trim()
   579→                if (optionsStr.startsWith('[') && optionsStr.endsWith(']')) {
   580→                  try {
   581→                    const parsed = JSON.parse(optionsStr)
   582→                    if (Array.isArray(parsed)) {
   583→                      options = parsed
   584→                    }
   585→                  } catch {
   586→                    options = optionsStr.split('|').map(opt => opt.trim())
   587→                  }
   588→                } else {
   589→                  options = optionsStr.split('|').map(opt => opt.trim())
   590→                }
   591→              }
   592→
   593→              // Create the question
   594→              const response = await fetch("/api/admin/question-groups", {
   595→                method: "POST",
   596→                headers: {
   597→                  "Content-Type": "application/json",
   598→                },
   599→                body: JSON.stringify({
   600→                  title: question.Title?.trim(),
   601→                  content: question.Content?.trim(),
   602→                  type: questionType,
   603→                  options: options,
   604→                  correctAnswer: question["Correct Answer"]?.toString(),
   605→                  explanation: question.Explanation?.trim() || "",
   606→                  difficulty: question.Difficulty?.toUpperCase() || "MEDIUM",
   607→                  groupId: selectedQuestionGroup
   608→                }),
   609→              })
   610→
   611→              if (!response.ok) {
   612→                const error = await response.json()
   613→                throw new Error(error.message || "Failed to create question")
   614→              }
   615→
   616→              return await response.json()
   617→            } catch (error) {
   618→              console.error(`Failed to import question ${index + 1}:`, error)
   619→              throw error
   620→            }
   621→          })
   622→
   623→          const importedQuestions = await Promise.all(importPromises)
   624→          
   625→          // Add imported questions to the assessment
   626→          const questionIds = importedQuestions.map(q => q.id)
   627→          const addResponse = await fetch(`/api/admin/assessment/${assessmentId}/questions`, {
   628→            method: "POST",
   629→            headers: {
   630→              "Content-Type": "application/json",
   631→            },
   632→            body: JSON.stringify({ questionIds }),
   633→          })
   634→
   635→          if (addResponse.ok) {
   636→            toast.success(`Successfully imported ${importedQuestions.length} questions and added to assessment`)
   637→            setIsImportSheetOpen(false)
   638→            setImportFile(null)
   639→            setSelectedQuestionGroup("")
   640→            fetchQuestions()
   641→            fetchAvailableQuestions()
   642→            setIsImporting(false)
   643→          } else {
   644→            toast.error("Questions were imported but failed to enroll to assessment")
   645→            setIsImporting(false)
   646→          }
   647→
   648→        } catch (error) {
   649→          console.error("Import error:", error)
   650→          toast.error(`Failed to import questions: ${error instanceof Error ? error.message : "Unknown error"}`)
