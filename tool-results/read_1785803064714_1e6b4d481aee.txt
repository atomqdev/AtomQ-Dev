    62→    62→const formatDateDDMMYYYY = (dateString: string) => {
    63→    63→  const date = new Date(dateString)
    64→    64→  const day = String(date.getDate()).padStart(2, '0')
    65→    65→  const month = String(date.getMonth() + 1).padStart(2, '0')
    66→    66→  const year = date.getFullYear()
    67→    67→  return `${day}/${month}/${year}`
    68→    68→}
