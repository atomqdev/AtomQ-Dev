   131→   131→        // Restore loaded questions

   133→   133→        for (let i = 0; i < existingProgress.quizData.questions.length; i++) {
   134→   134→          loadedSet.add(i)
   135→   135→        }
   136→   136→        setQuestionsLoaded(loadedSet)
   137→   137→        
   138→   138→        setIsRestoringProgress(false)

   143→   143→
   144→   144→      // Simulate progressive loading for better UX

   147→   147→          if (prev >= 90) {

   149→   149→            return prev
   150→   150→          }
   151→   151→          // return prev + Math.random() * 20
   152→   152→          return Math.min(prev + Math.random() * 20, 100)
   153→   153→        })
   154→   154→      }, 100)
   155→   155→
   156→   156→      const response = await fetch(`/api/user/quiz/${params.id}/attempt`)
   157→   157→      clearInterval(progressInterval)
   158→   158→      setLoadingProgress(100)
   159→   159→

   161→   161→        const data = await response.json()
   162→   162→        
   163→   163→        // Validate the response data
   164→   164→        if (!data.quiz || !data.quiz.questions || data.quiz.questions.length === 0) {
   165→   165→          console.error("Invalid quiz data received:", {
   166→   166→            hasQuiz: !!data.quiz,
   167→   167→            hasQuestions: !!data.quiz?.questions,
   168→   168→            questionsLength: data.quiz?.questions?.length,
   169→   169→            fullData: data

   172→   172→          router.push("/user/quiz")
   173→   173→          return
   174→   174→        }
   175→   175→        
   176→   176→        // Validate each question
   177→   177→        const invalidQuestions = data.quiz.questions.filter((q: any) => !q.id || !q.content)
   178→   178→        if (invalidQuestions.length > 0) {
   179→   179→          console.error("Found invalid questions:", invalidQuestions)
   180→   180→          toasts.error("Some quiz questions are corrupted")
   181→   181→          router.push("/user/quiz")
   182→   182→          return
   183→   183→        }
   184→   184→        
   185→   185→        // Initialize quiz in progress store
   186→   186→        startQuiz(params.id as string, data.attemptId, data.quiz, data.timeRemaining || 0)
   187→   187→        
   188→   188→        setQuiz(data.quiz)
   189→   189→        setAttemptId(data.attemptId)
   190→   190→        setTimeRemaining(data.timeRemaining || 0)
   191→   191→        setCanShowAnswers(data.quiz.showAnswers || false)
   192→   192→        
   193→   193→        // Preload first few questions for better performance
   194→   194→        const initialQuestionsToLoad = Math.min(5, data.quiz.questions.length)
   195→   195→        const loadedSet = new Set<number>()
   196→   196→        for (let i = 0; i < initialQuestionsToLoad; i++) {
   197→   197→          loadedSet.add(i)
   198→   198→        }
   199→   199→        setQuestionsLoaded(loadedSet)
   200→   200→        
   201→   201→        // Set initial answers if any exist
   202→   202→        if (data.answers) {
   203→   203→          setAnswers(data.answers)
   204→   204→          // Save initial answers to store
   205→   205→          Object.entries(data.answers).forEach(([questionId, answer]) => {
   206→   206→            saveAnswer(questionId, answer as string)
   207→   207→          })
   208→   208→        }
   209→   209→        
   210→   210→        // Lazy load remaining questions in background
   211→   211→        if (data.quiz.questions.length > initialQuestionsToLoad) {
   212→   212→          setTimeout(() => {
   213→   213→            const remainingSet = new Set(loadedSet)
   214→   214→            for (let i = initialQuestionsToLoad; i < data.quiz.questions.length; i++) {
   215→   215→              remainingSet.add(i)
   216→   216→            }
   217→   217→            setQuestionsLoaded(remainingSet)
   218→   218→          }, 1000)
   219→   219→        }
   220→   220→      } else {
   221→   221→        const error = await response.json()
   222→   222→        console.error("Failed to fetch quiz:", error)
   223→   223→        toasts.error(error.message || "Failed to load quiz")
   224→   224→        router.push("/user/quiz")
   225→   225→      }
   226→   226→    } catch (error) {
   227→   227→      console.error("Error in fetchQuiz:", error)
   228→   228→      toasts.error("Failed to load quiz")
   229→   229→      router.push("/user/quiz")
   230→   230→    } finally {
   231→   231→      // Add a small delay for smooth transition
   232→   232→      setTimeout(() => {
   233→   233→        setLoading(false)
   234→   234→      }, 300)
   235→   235→    }

   237→   237→
   238→   238→  useEffect(() => {
   239→   239→    if (session) {
   240→   240→      fetchQuiz()
   241→   241→    }
   242→   242→  }, [session, fetchQuiz])
   243→   243→
   244→   244→  // Timer

   248→   248→        const newTime = timeRemaining - 1

   256→   256→          // Auto-submit when time runs out

   263→   263→
   264→   264→              // Convert multi-select answers to pipe-separated string
   265→   265→              Object.keys(multiSelectAnswersRef.current).forEach(questionId => {
   266→   266→                const selectedOptions = multiSelectAnswersRef.current[questionId]

   268→   268→                  finalAnswers[questionId] = selectedOptions.join('|')

   271→   271→
   272→   272→              const response = await fetch(`/api/user/quiz/${params.id}/submit`, {
   273→   273→                method: "POST",
   274→   274→                headers: {
   275→   275→                  "Content-Type": "application/json",
   276→   276→                },
   277→   277→                body: JSON.stringify({
   278→   278→                  attemptId,
   279→   279→                  answers: finalAnswers,
   280→   280→                }),
   281→   281→              })
   282→   282→
   283→   283→              if (response.ok) {
   284→   284→                const result = await response.json()
   285→   285→                toasts.success("Time's up! Quiz submitted automatically.")
   286→   286→                // Clear persisted data after successful submission
   287→   287→                endQuiz()
   288→   288→                router.push(`/user/quiz/${params.id}/result?attemptId=${attemptId}`)
   289→   289→              } else {
   290→   290→                const error = await response.json()
   291→   291→                toasts.error(error.message || "Failed to submit quiz")
   292→   292→              }
   293→   293→            } catch (error) {
   294→   294→              toasts.error("Failed to submit quiz")
   295→   295→            } finally {
   296→   296→              setSubmitting(false)

   298→   298→          })()
   299→   299→        }
   300→   300→      }, 1000)
   301→   301→
   302→   302→      return () => clearInterval(timer)
   303→   303→    }
   304→   304→  }, [timeRemaining, submitting, attemptId, params.id, router, updateTimeRemaining, endQuiz])
   305→   305→
   306→   306→  // Auto-scroll pagination to active question
   307→   307→  useEffect(() => {
   308→   308→    if (paginationContainerRef.current && quiz) {
   309→   309→      const container = paginationContainerRef.current
   310→   310→      const activeButton = container.querySelector(`[data-question-index="${currentQuestionIndex}"]`) as HTMLElement
   311→   311→      
   312→   312→      if (activeButton) {
   313→   313→        const containerRect = container.getBoundingClientRect()
   314→   314→        const buttonRect = activeButton.getBoundingClientRect()
   315→   315→

   318→   318→

   321→   321→          left: container.scrollLeft + scrollLeft,
   322→   322→          behavior: 'smooth'

   325→   325→    }
   326→   326→  }, [currentQuestionIndex, quiz])
   327→   327→  useEffect(() => {

   332→   332→    document.addEventListener('fullscreenchange', handleFullscreenChange)

   334→   334→  }, [])
   335→   335→
   336→   336→  const toggleFullscreen = async () => {

   347→   347→
   348→   348→  const handleAnswerChange = (questionId: string, answer: string) => {

   360→   360→    // Save to persistence store

   363→   363→
   364→   364→  const handleMultiSelectAnswerChange = (questionId: string, option: string, checked: boolean) => {

   366→   366→    if (checkedAnswers.has(questionId)) {
   367→   367→      return
   368→   368→    }
   369→   369→
   370→   370→    // Calculate new answers first

   373→   373→
   374→   374→    if (checked) {

   376→   376→    } else {
   377→   377→      newAnswers = currentAnswers.filter(ans => ans !== option)
   378→   378→    }
   379→   379→
   380→   380→    // Update local state
   381→   381→    setMultiSelectAnswers(prev => ({
   382→   382→      ...prev,
   383→   383→      [questionId]: newAnswers

   386→   386→    // Save to persistence store (outside of setState callback)

   393→   393→    } else {

   397→   397→    }

   404→   404→  }

   409→   409→    }

   412→   412→  const goToQuestion = (index: number) => {

   416→   416→    
   417→   417→    // Load current question and nearby ones

   419→   419→      newLoadedSet.add(i)
   420→   420→    }
   421→   421→    
   422→   422→    setQuestionsLoaded(newLoadedSet)
   423→   423→    setCurrentQuestionIndex(index)

   425→   425→    // Save navigation to persistence store

   427→   427→  }
   428→   428→
   429→   429→  const handleSubmit = async () => {

   432→   432→    // Prepare answers for submission

   434→   434→    
   435→   435→    // Convert multi-select answers to pipe-separated string

   438→   438→      if (selectedOptions.length > 0) {
   439→   439→        finalAnswers[questionId] = selectedOptions.join('|')
   440→   440→      }
   441→   441→    })
   442→   442→
   443→   443→    setSubmitting(true)
   444→   444→    try {
   445→   445→      const response = await fetch(`/api/user/quiz/${params.id}/submit`, {
   446→   446→        method: "POST",
   447→   447→        headers: {
   448→   448→          "Content-Type": "application/json",
   449→   449→        },
   450→   450→        body: JSON.stringify({
   451→   451→          attemptId,
   452→   452→          answers: finalAnswers,
   453→   453→        }),
   454→   454→      })
   455→   455→
   456→   456→      if (response.ok) {
   457→   457→        const result = await response.json()
   458→   458→        toasts.success("Quiz submitted successfully!")
   459→   459→        // Clear persisted data after successful submission
   460→   460→        endQuiz()
   461→   461→        router.push(`/user/quiz/${params.id}/result?attemptId=${attemptId}`)
   462→   462→      } else {
   463→   463→        const error = await response.json()
   464→   464→        toasts.error(error.message || "Failed to submit quiz")
   465→   465→      }
   466→   466→    } catch (error) {
   467→   467→      toasts.error("Failed to submit quiz")
   468→   468→    } finally {
   469→   469→      setSubmitting(false)
   470→   470→    }
   471→   471→  }
   472→   472→
   473→   473→  const handleReportQuestion = async () => {
   474→   474→    if (!quiz || !session?.user?.id || !reportSuggestion.trim()) {
   475→   475→      toasts.error("Please provide a suggestion for the report")
   476→   476→      return
   477→   477→    }
   478→   478→
   479→   479→    setIsSubmittingReport(true)
   480→   480→    try {
   481→   481→      const response = await fetch("/api/user/question/report", {
   482→   482→        method: "POST",
   483→   483→        headers: {
   484→   484→          "Content-Type": "application/json",
   485→   485→        },
   486→   486→        body: JSON.stringify({
   487→   487→          questionId: quiz.questions[currentQuestionIndex].id,
   488→   488→          suggestion: reportSuggestion.trim(),
   489→   489→        }),
   490→   490→      })
   491→   491→
   492→   492→      const responseData = await response.json()
   493→   493→
   494→   494→      if (response.ok) {
   495→   495→        toasts.success("Question reported successfully!")
   496→   496→        setIsReportDialogOpen(false)
   497→   497→        setReportSuggestion("")
   498→   498→      } else {
   499→   499→        const errorMessage = responseData.error || responseData.details || "Failed to report question"
   500→   500→        toasts.error(errorMessage)
   501→   501→      }
   502→   502→    } catch (error) {
   503→   503→      toasts.error("Network error: Failed to report question")
   504→   504→    } finally {
