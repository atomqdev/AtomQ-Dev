    55→const formatDateDDMMYYYY = (dateString: string) => {
    56→  const date = new Date(dateString)
    57→  const day = String(date.getDate()).padStart(2, '0')
    58→  const month = String(date.getMonth() + 1).padStart(2, '0')
    59→  const year = date.getFullYear()
    60→  return `${day}/${month}/${year}`
    61→}
