   136→function SortableQuestion({
   137→  question,
   138→  onEdit,
   139→  onDelete,
   140→  onView
   141→}: {
   142→  question: Question
   143→  onEdit: (question: Question) => void
   144→  onDelete: (questionId: string) => void
   145→  onView: (question: Question) => void
   146→}) {
   147→  const {
   148→    attributes,
   149→    listeners,
   150→    setNodeRef,
   151→    transform,
   152→    transition,
   153→    isDragging,
   154→  } = useSortable({ id: question.id })
   155→
   156→  const style = {
   157→    transform: CSS.Transform.toString(transform),
   158→    transition,
   159→    opacity: isDragging ? 0.5 : 1,
   160→  }
   161→
   162→  return (
   163→    <TableRow ref={setNodeRef} style={style} {...attributes}>
   164→      <TableCell>
   165→        <div {...listeners} className="cursor-grab">
   166→          <GripVertical className="h-4 w-4" />
   167→        </div>
   168→      </TableCell>
   169→      <TableCell className="font-medium">{question.order}</TableCell>
   170→      <TableCell>
   171→        <div>
   172→          <div className="font-medium">{question.reference}</div>
   173→          {/* <div className="text-sm text-muted-foreground">
   174→            <RichTextDisplay content={question.title} />
   175→          </div> */}
   176→        </div>
   177→      </TableCell>
   178→      <TableCell>
   179→        <Badge variant={
   180→          question.type === QuestionType.MULTIPLE_CHOICE ? "default" :
   181→            question.type === QuestionType.MULTI_SELECT ? "secondary" :
   182→            question.type === QuestionType.TRUE_FALSE ? "outline" : "destructive"
   183→        }>
   184→          {question.type.replace('_', ' ')}
   185→          {question.type === QuestionType.MULTI_SELECT && (() => {
   186→            const sc = question.correctAnswer?.split('|').length || 0
   187→            return sc > 0 ? ` (Select ${sc})` : ''
   188→          })()}
   189→        </Badge>
   190→      </TableCell>
   191→      <TableCell>
   192→        <Badge variant={
   193→          question.difficulty === DifficultyLevel.EASY ? "default" :
   194→            question.difficulty === DifficultyLevel.MEDIUM ? "secondary" : "destructive"
   195→        }>
   196→          {question.difficulty}
   197→        </Badge>
   198→      </TableCell>
   199→      <TableCell>{question.points}</TableCell>
   200→      <TableCell>
   201→        <div className="flex gap-2">
   202→          <Button
   203→            variant="ghost"
   204→            size="sm"
   205→            onClick={() => onView(question)}
   206→          >
   207→            <Eye className="h-4 w-4" />
   208→          </Button>
   209→
   210→          <AlertDialog>
   211→            <AlertDialogTrigger asChild>
   212→              <Button
   213→                variant="ghost"
   214→                size="sm"
   215→                className="text-red-600"
   216→              >
   217→                <Trash2 className="h-4 w-4" />
   218→              </Button>
   219→            </AlertDialogTrigger>
   220→            <AlertDialogContent>
   221→              <AlertDialogHeader>
   222→                <AlertDialogTitle>Remove Question</AlertDialogTitle>
   223→                <AlertDialogDescription>
   224→                  Are you sure you want to remove this question from the quiz? This action cannot be undone.
   225→                </AlertDialogDescription>
   226→              </AlertDialogHeader>
   227→              <AlertDialogFooter>
   228→                <AlertDialogCancel>Cancel</AlertDialogCancel>
   229→                <AlertDialogAction onClick={() => onDelete(question.id)}>
   230→                  Remove
   231→                </AlertDialogAction>
   232→              </AlertDialogFooter>
   233→            </AlertDialogContent>
   234→          </AlertDialog>
   235→        </div>
   236→      </TableCell>
   237→    </TableRow>
   238→  )
   239→}

   425→  const handleExportQuestions = () => {
   426→    const csvContent = [
   427→      ["Reference", "Title", "Type", "Options", "Correct Answer", "Explanation", "Difficulty", "Points"],
   428→      ...questions.map(question => {
   429→        // Parse options from database (stored as JSON string) and convert to pipe-separated format

   431→        try {
   432→          const parsedOptions = Array.isArray(question.options)
   433→            ? question.options
   434→            : JSON.parse(question.options || "[]")
   435→          optionsString = parsedOptions.join("|")

   437→          console.warn("Failed to parse options for question:", question.reference, e)
   438→          optionsString = question.options?.toString() || ""
   439→        }
   440→
   441→        return [
   442→          question.reference,
   443→          question.title,
   444→          question.type,
   445→          optionsString,
   446→          question.correctAnswer,
   447→          question.explanation || "",
   448→          question.difficulty,

   522→    }

   531→    Papa.parse(importFile, {
   532→      header: true,
   533→      skipEmptyLines: true,
   534→      complete: async (results) => {
   535→        try {
   536→
   537→          // Filter out empty rows and validate required fields
   538→          const validQuestions = results.data.filter((row: any) => {
   539→            const hasReference = row.Reference && row.Reference.trim() !== ""
   540→            const hasTitle = row.Title && row.Title.trim() !== ""
   541→            const hasType = row.Type && row.Type.trim() !== ""
   542→            const hasOptions = row.Options && row.Options.trim() !== ""
   543→            const hasCorrectAnswer = row["Correct Answer"] && row["Correct Answer"].trim() !== ""
   544→
   545→            return hasReference && hasTitle && hasType && hasOptions && hasCorrectAnswer
   546→          })
   547→
   548→
   549→          if (validQuestions.length === 0) {
   550→            toast.error("No valid questions found in CSV file. Please ensure all required fields are filled.")
   551→            return
   552→          }
   553→
   554→          // Create questions in the selected group
   555→          const importPromises = validQuestions.map(async (question: any, index: number) => {
   556→            try {
   557→
   558→              // Normalize question type
   559→              let questionType = question.Type?.toString().toUpperCase().trim()
   560→              if (!Object.values(QuestionType).includes(questionType as QuestionType)) {
   561→                questionType = QuestionType.MULTIPLE_CHOICE
   562→                console.warn(`Invalid question type "${question.Type}", defaulting to MULTIPLE_CHOICE`)
   563→              }
   564→
   565→              // Parse options
   566→              let options = []
   567→              if (question.Options) {
   568→                const optionsStr = question.Options.toString().trim()
   569→                if (optionsStr.startsWith('[') && optionsStr.endsWith(']')) {
   570→                  try {
   571→                    const parsed = JSON.parse(optionsStr)
   572→                    if (Array.isArray(parsed)) {
   573→                      options = parsed
   574→                    }
   575→                  } catch {
   576→                    options = optionsStr.split('|').map(opt => opt.trim())
   577→                  }
   578→                } else {
   579→                  options = optionsStr.split('|').map(opt => opt.trim())
   580→                }
   581→              }
   582→
   583→              // Create the question
   584→              const response = await fetch("/api/admin/question-groups", {
   585→                method: "POST",
   586→                headers: {
   587→                  "Content-Type": "application/json",
   588→                },
   589→                body: JSON.stringify({
   590→                  reference: question.Reference?.trim(),
   591→                  title: question.Title?.trim(),
   592→                  type: questionType,
   593→                  options: options,
   594→                  correctAnswer: question["Correct Answer"]?.toString(),
   595→                  explanation: question.Explanation?.trim() || "",
   596→                  difficulty: question.Difficulty?.toUpperCase() || "MEDIUM",
   597→                  groupId: selectedQuestionGroup
   598→                }),
   599→              })
   600→
   601→              if (!response.ok) {
   602→                const error = await response.json()
   603→                throw new Error(error.message || "Failed to create question")
   604→              }
   605→
   606→              return await response.json()
   607→            } catch (error) {
   608→              console.error(`Failed to import question ${index + 1}:`, error)
   609→              throw error
   610→            }
   611→          })
   612→
   613→          const importedQuestions = await Promise.all(importPromises)
   614→          
   615→          // Add imported questions to the quiz
   616→          const questionIds = importedQuestions.map(q => q.id)
   617→          const addResponse = await fetch(`/api/admin/quiz/${quizId}/questions`, {
   618→            method: "POST",
   619→            headers: {
   620→              "Content-Type": "application/json",
   621→            },
   622→            body: JSON.stringify({ questionIds }),
   623→          })
   624→
   625→          if (addResponse.ok) {
   626→            toast.success(`Successfully imported ${importedQuestions.length} questions and added to quiz`)
   627→            setIsImportSheetOpen(false)
   628→            setImportFile(null)
   629→            setSelectedQuestionGroup("")
   630→            fetchQuestions()
   631→            fetchAvailableQuestions()
   632→            setIsImporting(false)
   633→          } else {
   634→            toast.error("Questions were imported but failed to enroll to quiz")
   635→            setIsImporting(false)
   636→          }
   637→
   638→        } catch (error) {
   639→          console.error("Import error:", error)
   640→          toast.error(`Failed to import questions: ${error instanceof Error ? error.message : "Unknown error"}`)
   641→          setIsImporting(false)
   642→        }
   643→      },
   644→      error: (error) => {
   645→        console.error("CSV parsing error:", error)
   646→        toast.error("Failed to parse CSV file")
   647→        setIsImporting(false)
   648→      }
   649→    })
   650→  }
   651→
