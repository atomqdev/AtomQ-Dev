    63→  const params = useParams()

    67→  const [quiz, setQuiz] = useState<Quiz | null>(null)

    69→  const [currentQuestionIndex, setCurrentQuestionIndex] = useState(0)
    70→  const [answers, setAnswers] = useState<Record<string, string>>({})
    71→  const [multiSelectAnswers, setMultiSelectAnswers] = useState<Record<string, string[]>>({})

    73→  const [loading, setLoading] = useState(true)
    74→  const [submitting, setSubmitting] = useState(false)

    77→  const [canShowAnswers, setCanShowAnswers] = useState(false)
    78→  const [checkedAnswers, setCheckedAnswers] = useState<Set<string>>(new Set())
    79→  const [loadingProgress, setLoadingProgress] = useState(0)
    80→  const [questionsLoaded, setQuestionsLoaded] = useState<Set<number>>(new Set())
    81→  const [isRestoringProgress, setIsRestoringProgress] = useState(false)
    82→  const [isReportDialogOpen, setIsReportDialogOpen] = useState(false)
    83→  const [reportSuggestion, setReportSuggestion] = useState("")
    84→  const [isSubmittingReport, setIsSubmittingReport] = useState(false)
    85→  const paginationContainerRef = useRef<HTMLDivElement>(null)
    86→  const answersRef = useRef<Record<string, string>>(answers)
    87→  const multiSelectAnswersRef = useRef<Record<string, string[]>>(multiSelectAnswers)
    88→
    89→  // Update refs when state changes
    90→  useEffect(() => {
    91→    answersRef.current = answers
    92→  }, [answers])
    93→
    94→  useEffect(() => {
    95→    multiSelectAnswersRef.current = multiSelectAnswers
    96→  }, [multiSelectAnswers])
    97→
    98→  // Quiz progress store
    99→  const {
   100→    startQuiz,
   101→    updateProgress,
   102→    saveAnswer,
   103→    saveMultiSelectAnswer,
   104→    navigateToQuestion,
   105→    updateTimeRemaining,
   106→    endQuiz,
   107→    getProgress,
   108→    isQuizInProgress,
   109→    getCurrentQuestion,
   110→    getAnswer,
   111→    getMultiSelectAnswer
   112→  } = useQuizProgressStore()
   113→
   114→  const fetchQuiz = useCallback(async () => {
   115→    try {
   116→      // Check if there's existing progress in localStorage first
   117→      const existingProgress = getProgress(params.id as string)
   118→      const attemptFromUrl = searchParams.get('attempt')
   119→      
   120→      if (existingProgress && existingProgress.attemptId === attemptFromUrl) {
   121→        // Restore progress from localStorage
   122→        setIsRestoringProgress(true)
   123→        setQuiz(existingProgress.quizData)
   124→        setAttemptId(existingProgress.attemptId)
   125→        setCurrentQuestionIndex(existingProgress.currentQuestionIndex)
   126→        setAnswers(existingProgress.answers)
   127→        setMultiSelectAnswers(existingProgress.multiSelectAnswers)
   128→        setTimeRemaining(existingProgress.timeRemaining)
   129→        setCanShowAnswers(existingProgress.quizData.showAnswers || false)
   130→        
   131→        // Restore loaded questions
   132→        const loadedSet = new Set<number>()
   133→        for (let i = 0; i < existingProgress.quizData.questions.length; i++) {
   134→          loadedSet.add(i)
   135→        }
   136→        setQuestionsLoaded(loadedSet)
   137→        
   138→        setIsRestoringProgress(false)
   139→        setLoading(false)
   140→        toasts.success("Quiz progress restored")
   141→        return
   142→      }
   143→
   144→      // Simulate progressive loading for better UX
   145→      const progressInterval = setInterval(() => {
   146→        setLoadingProgress(prev => {
   147→          if (prev >= 90) {
   148→            clearInterval(progressInterval)
   149→            return prev
   150→          }
   151→          // return prev + Math.random() * 20
   152→          return Math.min(prev + Math.random() * 20, 100)
   153→        })
   154→      }, 100)
   155→
   156→      const response = await fetch(`/api/user/quiz/${params.id}/attempt`)
   157→      clearInterval(progressInterval)
   158→      setLoadingProgress(100)
   159→      
   160→      if (response.ok) {
   161→        const data = await response.json()
   162→        
   163→        // Validate the response data
   164→        if (!data.quiz || !data.quiz.questions || data.quiz.questions.length === 0) {
   165→          console.error("Invalid quiz data received:", {
   166→            hasQuiz: !!data.quiz,
   167→            hasQuestions: !!data.quiz?.questions,
   168→            questionsLength: data.quiz?.questions?.length,
   169→            fullData: data
   170→          })
   171→          toasts.error("Quiz data is incomplete or corrupted")
   172→          router.push("/user/quiz")
   173→          return
   174→        }
   175→        
   176→        // Validate each question
   177→        const invalidQuestions = data.quiz.questions.filter((q: any) => !q.id || !q.title)
   178→        if (invalidQuestions.length > 0) {
   179→          console.error("Found invalid questions:", invalidQuestions)
   180→          toasts.error("Some quiz questions are corrupted")
   181→          router.push("/user/quiz")
   182→          return
   183→        }
   184→        
   185→        // Initialize quiz in progress store
   186→        startQuiz(params.id as string, data.attemptId, data.quiz, data.timeRemaining || 0)
   187→        
   188→        setQuiz(data.quiz)
   189→        setAttemptId(data.attemptId)
   190→        setTimeRemaining(data.timeRemaining || 0)
   191→        setCanShowAnswers(data.quiz.showAnswers || false)
   192→        
   193→        // Preload first few questions for better performance
   194→        const initialQuestionsToLoad = Math.min(5, data.quiz.questions.length)
   195→        const loadedSet = new Set<number>()
   196→        for (let i = 0; i < initialQuestionsToLoad; i++) {
   197→          loadedSet.add(i)
   198→        }
   199→        setQuestionsLoaded(loadedSet)
   200→        
   201→        // Set initial answers if any exist
   202→        if (data.answers) {
   203→          setAnswers(data.answers)
   204→          // Save initial answers to store
   205→          Object.entries(data.answers).forEach(([questionId, answer]) => {
   206→            saveAnswer(questionId, answer as string)
   207→          })
   208→        }
   209→        
   210→        // Lazy load remaining questions in background
   211→        if (data.quiz.questions.length > initialQuestionsToLoad) {
   212→          setTimeout(() => {
   213→            const remainingSet = new Set(loadedSet)
   214→            for (let i = initialQuestionsToLoad; i < data.quiz.questions.length; i++) {
   215→              remainingSet.add(i)
   216→            }
   217→            setQuestionsLoaded(remainingSet)
   218→          }, 1000)
   219→        }
   220→      } else {
   221→        const error = await response.json()
   222→        console.error("Failed to fetch quiz:", error)
   223→        toasts.error(error.message || "Failed to load quiz")
   224→        router.push("/user/quiz")
   225→      }
   226→    } catch (error) {
   227→      console.error("Error in fetchQuiz:", error)
   228→      toasts.error("Failed to load quiz")
   229→      router.push("/user/quiz")
   230→    } finally {
   231→      // Add a small delay for smooth transition
   232→      setTimeout(() => {
   233→        setLoading(false)
   234→      }, 300)
   235→    }
   236→  }, [params.id, router, searchParams, startQuiz, saveAnswer, getProgress])
   237→
   238→  useEffect(() => {
   239→    if (session) {
   240→      fetchQuiz()
   241→    }
   242→  }, [session, fetchQuiz])
   243→
   244→  // Timer
   245→  useEffect(() => {
   246→    if (timeRemaining > 0 && !submitting) {
   247→      const timer = setInterval(() => {
   248→        const newTime = timeRemaining - 1
   249→        // Update local state first
   250→        setTimeRemaining(newTime)
   251→
   252→        // Then update persistence store (outside of setState callback)
   253→        updateTimeRemaining(newTime)
   254→
   255→        if (newTime <= 0) {
   256→          // Auto-submit when time runs out
   257→          ;(async () => {
   258→            try {
   259→              setSubmitting(true)
   260→
   261→              // Prepare answers for submission using refs to get latest values
   262→              const finalAnswers: Record<string, string> = { ...answersRef.current }
   263→
   264→              // Convert multi-select answers to pipe-separated string
   265→              Object.keys(multiSelectAnswersRef.current).forEach(questionId => {
   266→                const selectedOptions = multiSelectAnswersRef.current[questionId]
   267→                if (selectedOptions.length > 0) {
   268→                  finalAnswers[questionId] = selectedOptions.join('|')
   269→                }
   270→              })
   271→
   272→              const response = await fetch(`/api/user/quiz/${params.id}/submit`, {
   273→                method: "POST",
   274→                headers: {
   275→                  "Content-Type": "application/json",
   276→                },
   277→                body: JSON.stringify({
   278→                  attemptId,
   279→                  answers: finalAnswers,
   280→                }),
   281→              })
   282→
   283→              if (response.ok) {
   284→                const result = await response.json()
   285→                toasts.success("Time's up! Quiz submitted automatically.")
   286→                // Clear persisted data after successful submission
   287→                endQuiz()
   288→                router.push(`/user/quiz/${params.id}/result?attemptId=${attemptId}`)
   289→              } else {
   290→                const error = await response.json()
   291→                toasts.error(error.message || "Failed to submit quiz")
   292→              }
   293→            } catch (error) {
   294→              toasts.error("Failed to submit quiz")
   295→            } finally {
   296→              setSubmitting(false)
   297→            }
   298→          })()
   299→        }
   300→      }, 1000)
   301→
   302→      return () => clearInterval(timer)
   303→    }
   304→  }, [timeRemaining, submitting, attemptId, params.id, router, updateTimeRemaining, endQuiz])
   305→
   306→  // Auto-scroll pagination to active question
   307→  useEffect(() => {
   308→    if (paginationContainerRef.current && quiz) {
   309→      const container = paginationContainerRef.current
   310→      const activeButton = container.querySelector(`[data-question-index="${currentQuestionIndex}"]`) as HTMLElement
   311→      
   312→      if (activeButton) {
   313→        const containerRect = container.getBoundingClientRect()
   314→        const buttonRect = activeButton.getBoundingClientRect()
   315→        
   316→        // Calculate the scroll position to center the active button
   317→        const scrollLeft = buttonRect.left - containerRect.left - containerRect.width / 2 + buttonRect.width / 2
   318→        
   319→        // Smooth scroll to the active button
   320→        container.scrollTo({
   321→          left: container.scrollLeft + scrollLeft,
   322→          behavior: 'smooth'
   323→        })
   324→      }
   325→    }
   326→  }, [currentQuestionIndex, quiz])
   327→  useEffect(() => {
   328→    const handleFullscreenChange = () => {
   329→      setIsFullscreen(!!document.fullscreenElement)
   330→    }
   331→
   332→    document.addEventListener('fullscreenchange', handleFullscreenChange)
   333→    return () => document.removeEventListener('fullscreenchange', handleFullscreenChange)
   334→  }, [])
   335→
   336→  const toggleFullscreen = async () => {
   337→    try {
   338→      if (!document.fullscreenElement) {
   339→        await document.documentElement.requestFullscreen()
   340→      } else {
   341→        await document.exitFullscreen()
   342→      }
   343→    } catch (error) {
   344→      console.error('Error toggling fullscreen:', error)
   345→    }
   346→  }
   347→
   348→  const handleAnswerChange = (questionId: string, answer: string) => {
   349→    // Prevent changing answer if it has been checked
   350→    if (checkedAnswers.has(questionId)) {
   351→      return
   352→    }
   353→    
   354→    // Update local state
   355→    setAnswers(prev => ({
   356→      ...prev,
   357→      [questionId]: answer
   358→    }))
   359→    
   360→    // Save to persistence store
   361→    saveAnswer(questionId, answer)
   362→  }
   363→
   364→  const handleMultiSelectAnswerChange = (questionId: string, option: string, checked: boolean) => {
   365→    // Prevent changing answer if it has been checked
   366→    if (checkedAnswers.has(questionId)) {
   367→      return
   368→    }
   369→
   370→    // Calculate new answers first
   371→    const currentAnswers = multiSelectAnswers[questionId] || []
   372→    let newAnswers: string[]
   373→
   374→    if (checked) {
   375→      newAnswers = [...currentAnswers, option]
   376→    } else {
   377→      newAnswers = currentAnswers.filter(ans => ans !== option)
   378→    }
   379→
   380→    // Update local state
   381→    setMultiSelectAnswers(prev => ({
   382→      ...prev,
   383→      [questionId]: newAnswers
   384→    }))
   385→
   386→    // Save to persistence store (outside of setState callback)
   387→    saveMultiSelectAnswer(questionId, newAnswers)
   388→  }
   389→
   390→  const handleCheckAnswer = (questionId: string) => {
   391→    if (showAnswer === questionId) {
   392→      setShowAnswer(null)
   393→    } else {
   394→      setShowAnswer(questionId)
   395→      // Add this question to checked answers set to lock it
   396→      setCheckedAnswers(prev => new Set(prev).add(questionId))
   397→    }
   398→  }
   399→
   400→  const nextQuestion = () => {
   401→    if (currentQuestionIndex < (quiz?.questions.length || 0) - 1) {
   402→      goToQuestion(currentQuestionIndex + 1)
   403→    }
   404→  }
   405→
   406→  const previousQuestion = () => {
   407→    if (currentQuestionIndex > 0) {
   408→      goToQuestion(currentQuestionIndex - 1)
   409→    }
   410→  }
   411→
   412→  const goToQuestion = (index: number) => {
   413→    // Preload nearby questions for better performance
   414→    const preloadRange = 2
   415→    const newLoadedSet = new Set(questionsLoaded)
   416→    
   417→    // Load current question and nearby ones
   418→    for (let i = Math.max(0, index - preloadRange); i <= Math.min((quiz?.questions.length || 0) - 1, index + preloadRange); i++) {
   419→      newLoadedSet.add(i)
   420→    }
   421→    
   422→    setQuestionsLoaded(newLoadedSet)
   423→    setCurrentQuestionIndex(index)
   424→    
   425→    // Save navigation to persistence store
   426→    navigateToQuestion(index)
   427→  }
   428→
   429→  const handleSubmit = async () => {
   430→    if (submitting) return
   431→
   432→    // Prepare answers for submission
   433→    const finalAnswers: Record<string, string> = { ...answers }
   434→    
   435→    // Convert multi-select answers to pipe-separated string
   436→    Object.keys(multiSelectAnswers).forEach(questionId => {
   437→      const selectedOptions = multiSelectAnswers[questionId]
   438→      if (selectedOptions.length > 0) {
   439→        finalAnswers[questionId] = selectedOptions.join('|')
   440→      }
   441→    })
   442→
   443→    setSubmitting(true)
   444→    try {
   445→      const response = await fetch(`/api/user/quiz/${params.id}/submit`, {
   446→        method: "POST",
   447→        headers: {
   448→          "Content-Type": "application/json",
   449→        },
   450→        body: JSON.stringify({
   451→          attemptId,
   452→          answers: finalAnswers,
   453→        }),
   454→      })
   455→
   456→      if (response.ok) {
   457→        const result = await response.json()
   458→        toasts.success("Quiz submitted successfully!")
   459→        // Clear persisted data after successful submission
   460→        endQuiz()
   461→        router.push(`/user/quiz/${params.id}/result?attemptId=${attemptId}`)
   462→      } else {
   463→        const error = await response.json()
   464→        toasts.error(error.message || "Failed to submit quiz")
   465→      }
   466→    } catch (error) {
   467→      toasts.error("Failed to submit quiz")
   468→    } finally {
   469→      setSubmitting(false)
   470→    }
   471→  }
   472→
   473→  const handleReportQuestion = async () => {
   474→    if (!quiz || !session?.user?.id || !reportSuggestion.trim()) {
   475→      toasts.error("Please provide a suggestion for the report")
   476→      return
   477→    }
   478→
   479→    setIsSubmittingReport(true)
   480→    try {
   481→      const response = await fetch("/api/user/question/report", {
   482→        method: "POST",
   483→        headers: {
   484→          "Content-Type": "application/json",
   485→        },
   486→        body: JSON.stringify({
   487→          questionId: quiz.questions[currentQuestionIndex].id,
   488→          suggestion: reportSuggestion.trim(),
   489→        }),
   490→      })
   491→
   492→      const responseData = await response.json()
   493→
   494→      if (response.ok) {
   495→        toasts.success("Question reported successfully!")
   496→        setIsReportDialogOpen(false)
   497→        setReportSuggestion("")
   498→      } else {
   499→        const errorMessage = responseData.error || responseData.details || "Failed to report question"
   500→        toasts.error(errorMessage)
   501→      }
   502→    } catch (error) {
   503→      toasts.error("Network error: Failed to report question")
   504→    } finally {
   505→      setIsSubmittingReport(false)
   506→    }
   507→  }
   508→
   509→  const formatTime = (seconds: number) => {
   510→    const minutes = Math.floor(seconds / 60)
   511→    const remainingSeconds = seconds % 60
   512→    return `${minutes}:${remainingSeconds.toString().padStart(2, '0')}`

   514→
   515→  const scrollToActiveQuestion = useCallback(() => {
   516→    if (paginationContainerRef.current) {
   517→      const container = paginationContainerRef.current
   518→      const activeButton = container.querySelector(`[data-question-index="${currentQuestionIndex}"]`) as HTMLElement
   519→      
   520→      if (activeButton) {
   521→        const containerRect = container.getBoundingClientRect()
   522→        const buttonRect = activeButton.getBoundingClientRect()
   523→        
   524→        // Calculate the scroll position to center the active button
   525→        const scrollLeft = buttonRect.left - containerRect.left + container.scrollLeft - (containerRect.width / 2) + (buttonRect.width / 2)
   526→        
   527→        // Smooth scroll to the active button
   528→        container.scrollTo({
   529→          left: scrollLeft,
   530→          behavior: 'smooth'
   531→        })
   532→      }
   533→    }
   534→  }, [currentQuestionIndex])
   535→
   536→  // Scroll to active question when currentQuestionIndex changes
   537→  useEffect(() => {
   538→    scrollToActiveQuestion()
   539→  }, [currentQuestionIndex, scrollToActiveQuestion])
   540→
   541→  const formatQuestionType = (type: QuestionType) => {
   542→    switch (type) {
   543→      case QuestionType.MULTIPLE_CHOICE:
   544→        return "Multiple Choice"
   545→      case QuestionType.TRUE_FALSE:
   546→        return "True/False"
   547→      case QuestionType.FILL_IN_BLANK:
   548→        return "Fill in Blank"
   549→      case QuestionType.MULTI_SELECT:
   550→        return "Multi-Select"
   551→      default:
   552→        return "Unknown"
   553→    }
   554→  }
   555→
   556→  const getQuestionTypeColor = (type: QuestionType) => {
   557→    switch (type) {
   558→      case QuestionType.MULTIPLE_CHOICE:
   559→        return "text-blue-600 bg-blue-50 dark:text-blue-400 dark:bg-blue-900/30"
   560→      case QuestionType.TRUE_FALSE:
   561→        return "text-green-600 bg-green-50 dark:text-green-400 dark:bg-green-900/30"
   562→      case QuestionType.FILL_IN_BLANK:
   563→        return "text-primary bg-primary/10 dark:text-primary dark:bg-primary/20"
   564→      case QuestionType.MULTI_SELECT:
   565→        return "text-orange-600 bg-orange-50 dark:text-orange-400 dark:bg-orange-900/30"
   566→      default:
   567→        return "text-gray-600 bg-gray-50 dark:text-gray-400 dark:bg-gray-900/30"
   568→    }
   569→  }
   570→
   571→  const isQuestionAnswered = (question: Question) => {
   572→    if (question.type === QuestionType.MULTI_SELECT) {
   573→      return multiSelectAnswers[question.id]?.length > 0
   574→    }
   575→    return answers[question.id]
   576→  }
   577→
   578→  const isQuestionLoaded = (questionIndex: number) => {
   579→    return questionsLoaded.has(questionIndex)
   580→  }
   581→
   582→  const getTimeColor = () => {
   583→    const percentage = (timeRemaining / ((quiz?.timeLimit || 0) * 60 || 1)) * 100
   584→    if (percentage > 50) return "text-green-500"
   585→    if (percentage > 20) return "text-yellow-500"
   586→    return "text-red-500"
   587→  }
   588→
   589→  if (loading) {
   590→    return (
   591→      <div className="min-h-screen bg-background dark:bg-background flex items-center justify-center">
   592→        <motion.div
   593→          initial={{ opacity: 0, scale: 0.9 }}
   594→          animate={{ opacity: 1, scale: 1 }}
   595→          className="text-center"
   596→        >
   597→          <div className="mb-6">
   598→            <HexagonLoader size={120} />
   599→          </div>
   600→          <motion.div
   601→            initial={{ opacity: 0, y: 20 }}
   602→            animate={{ opacity: 1, y: 0 }}
   603→            transition={{ delay: 0.3 }}
   604→            className="space-y-4"
   605→          >
   606→            <p className="text-xl font-semibold text-primary">Preparing your quiz...</p>
   607→            <p className="text-muted-foreground">Loading questions and setting up your experience</p>
   608→
