    68→    68→// Debounce hook to prevent too many API calls
    69→    69→function useDebounce<T>(value: T, delay: number): T {
    70→    70→  const [debouncedValue, setDebouncedValue] = useState<T>(value)
    71→    71→
    72→    72→  useEffect(() => {
    73→    73→    const handler = setTimeout(() => {
    74→    74→      setDebouncedValue(value)
    75→    75→    }, delay)
    76→    76→
    77→    77→    return () => {
    78→    78→      clearTimeout(handler)
    79→    79→    }
    80→    80→  }, [value, delay])
    81→    81→
    82→    82→  return debouncedValue
    83→    83→}
