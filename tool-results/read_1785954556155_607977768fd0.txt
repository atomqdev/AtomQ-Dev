   534→  const handleImportJSON = async (event: React.ChangeEvent<HTMLInputElement>) => {
   535→    const file = event.target.files?.[0]
   536→    if (!file) return
   537→
   538→    // Reset the input so the same file can be selected again
   539→    event.target.value = ""
   540→
   541→    if (!file.name.endsWith('.json')) {
   542→      toasts.error("Please select a JSON file")
   543→      return
   544→    }
   545→
   546→    setImportLoading(true)
   547→    try {
   548→      const text = await file.text()
   549→      let parsedData: unknown
   550→      try {
   551→        parsedData = JSON.parse(text)
   552→      } catch {
   553→        toasts.error("Invalid JSON file. Please check the file format.")
   554→        return
   555→      }
   556→
   557→      // Support both a raw array and an object with an `importData`/`quizzes` array
   558→      let importArray: any[] = []
   559→      if (Array.isArray(parsedData)) {
   560→        importArray = parsedData
   561→      } else if (parsedData && typeof parsedData === 'object') {
   562→        const obj = parsedData as Record<string, unknown>
   563→        if (Array.isArray(obj.importData)) {
   564→          importArray = obj.importData as any[]
   565→        } else if (Array.isArray(obj.quizzes)) {
   566→          importArray = obj.quizzes as any[]
   567→        }
   568→      }
   569→
   570→      if (importArray.length === 0) {
   571→        toasts.error("No quiz records found in the JSON file")
   572→        return
   573→      }
   574→
   575→      const response = await fetch("/api/admin/quiz", {
   576→        method: "POST",
   577→        headers: { "Content-Type": "application/json" },
   578→        body: JSON.stringify({ importData: importArray, groupId: id }),
   579→      })
   580→
   581→      if (response.ok) {
   582→        const result = await response.json()
   583→        toasts.success(result.message || `Import completed: ${result.successCount} created, ${result.failureCount} failed`)
   584→        await fetchQuizzes()
   585→        fetchGroup()
   586→      } else if (response.status === 401) {
   587→        toasts.error("Session expired. Please log in again.")
   588→        router.push('/')
   589→      } else {
   590→        const errorData = await response.json().catch(() => ({}))
   591→        toasts.error(errorData.message || "Failed to import quizzes")
   592→      }
   593→    } catch (error) {
   594→      console.error("Error importing quizzes:", error)
   595→      toasts.networkError()
   596→    } finally {
   597→      setImportLoading(false)
   598→    }
   599→  }
