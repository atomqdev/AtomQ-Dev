    75→// Helper function to generate a 6-digit alphanumeric access key (format: 1a-2b-3c)
    76→const generateAccessKey = () => {
    77→  const generatePart = () => {
    78→    const num = Math.floor(Math.random() * 10).toString()
    79→    const char = String.fromCharCode(97 + Math.floor(Math.random() * 26)) // a-z
    80→    return num + char
    81→  }
    82→  return `${generatePart()}-${generatePart()}-${generatePart()}`
    83→}

    85→// Helper function to format dates in dd/mm/yyyy format
    86→const formatDateDDMMYYYY = (dateString: string) => {
    87→  const date = new Date(dateString)
    88→  const day = String(date.getDate()).padStart(2, '0')
    89→  const month = String(date.getMonth() + 1).padStart(2, '0')
    90→  const year = date.getFullYear()
    91→  return `${day}/${month}/${year}`
    92→}
    93→
    94→// Helper function to format dates with time (12-hour format)
    95→const formatDateTime = (dateString: string) => {
    96→  const date = new Date(dateString)
    97→  const day = String(date.getDate()).padStart(2, '0')
    98→  const month = String(date.getMonth() + 1).padStart(2, '0')
    99→  const year = date.getFullYear()
   100→  const hours24 = date.getHours()
   101→  const hours = hours24 % 12 || 12 // Convert to 12-hour format
   102→  const minutes = String(date.getMinutes()).padStart(2, '0')
   103→  const ampm = hours24 >= 12 ? 'PM' : 'AM'
   104→  return `${day}/${month}/${year} ${hours}:${minutes} ${ampm}`
   105→}
