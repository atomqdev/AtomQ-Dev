   136→function SortableQuestion({
   137→  question,
   138→  onEdit,
   139→  onDelete,
   140→  onView
   141→}: {
   142→  question: Question
   143→  onEdit: (question: Question) => void
   144→  onDelete: (questionId: string) => void
   145→  onView: (question: Question) => void
   146→}) {
   147→  const {
   148→    attributes,
   149→    listeners,
   150→    setNodeRef,
   151→    transform,
   152→    transition,
   153→    isDragging,
   154→  } = useSortable({ id: question.id })
   155→
   156→  const style = {
   157→    transform: CSS.Transform.toString(transform),
   158→    transition,
   159→    opacity: isDragging ? 0.5 : 1,
   160→  }
   161→
   162→  return (
   163→    <TableRow ref={setNodeRef} style={style} {...attributes}>
   164→      <TableCell>
   165→        <div {...listeners} className="cursor-grab">
   166→          <GripVertical className="h-4 w-4" />
   167→        </div>
   168→      </TableCell>
   169→      <TableCell className="font-medium">{question.order}</TableCell>
   170→      <TableCell>
   171→        <div>
   172→          <div className="font-medium">{question.title}</div>
   173→          {/* <div className="text-sm text-muted-foreground">
   174→            <RichTextDisplay content={question.content} />
   175→          </div> */}
   176→        </div>
   177→      </TableCell>
   178→      <TableCell>
   179→        <Badge variant={
   180→          question.type === QuestionType.MULTIPLE_CHOICE ? "default" :
   181→            question.type === QuestionType.MULTI_SELECT ? "secondary" :
   182→            question.type === QuestionType.TRUE_FALSE ? "outline" : "destructive"
   183→        }>
   184→          {question.type.replace('_', ' ')}
   185→        </Badge>
   186→      </TableCell>
   187→      <TableCell>
   188→        <Badge variant={
   189→          question.difficulty === DifficultyLevel.EASY ? "default" :
   190→            question.difficulty === DifficultyLevel.MEDIUM ? "secondary" : "destructive"
   191→        }>
   192→          {question.difficulty}
   193→        </Badge>
   194→      </TableCell>
   195→      <TableCell>{question.points}</TableCell>
   196→      <TableCell>
   197→        <div className="flex gap-2">
   198→          <Button
   199→            variant="ghost"
   200→            size="sm"
   201→            onClick={() => onView(question)}
   202→          >
   203→            <Eye className="h-4 w-4" />
   204→          </Button>
   205→
   206→          <AlertDialog>
   207→            <AlertDialogTrigger asChild>
   208→              <Button
   209→                variant="ghost"
   210→                size="sm"
   211→                className="text-red-600"
   212→              >
   213→                <Trash2 className="h-4 w-4" />
   214→              </Button>
   215→            </AlertDialogTrigger>
   216→            <AlertDialogContent>
   217→              <AlertDialogHeader>
   218→                <AlertDialogTitle>Remove Question</AlertDialogTitle>
   219→                <AlertDialogDescription>
   220→                  Are you sure you want to remove this question from the quiz? This action cannot be undone.
   221→                </AlertDialogDescription>
   222→              </AlertDialogHeader>
   223→              <AlertDialogFooter>
   224→                <AlertDialogCancel>Cancel</AlertDialogCancel>
   225→                <AlertDialogAction onClick={() => onDelete(question.id)}>
   226→                  Remove
   227→                </AlertDialogAction>
   228→              </AlertDialogFooter>
   229→            </AlertDialogContent>
   230→          </AlertDialog>
   231→        </div>
   232→      </TableCell>
   233→    </TableRow>
   234→  )
   235→}

   303→        const data = await response.json()

   307→      toast.error("Failed to fetch questions")

   330→  const fetchQuestionGroups = async () => {

   334→        const data = await response.json()
   335→        setQuestionGroups(data)

   338→      console.error("Failed to fetch question groups:", error)
   339→    }

   341→
   342→  const handleDragEnd = async (event: DragEndEvent) => {

   396→        method: "POST",
   397→        headers: {
   398→          "Content-Type": "application/json",

   414→        toast.error("Failed to enroll questions")

   417→      toast.error("Failed to enroll questions")
   418→    }
   419→  }
   420→
   421→  const handleExportQuestions = () => {
   422→    const csvContent = [
   423→      ["Title", "Content", "Type", "Options", "Correct Answer", "Explanation", "Difficulty", "Points"],
   424→      ...questions.map(question => {
   425→        // Parse options from database (stored as JSON string) and convert to pipe-separated format
   426→        let optionsString = ""
   427→        try {
   428→          const parsedOptions = Array.isArray(question.options)
   429→            ? question.options
   430→            : JSON.parse(question.options || "[]")
   431→          optionsString = parsedOptions.join("|")
   432→        } catch (e) {
   433→          console.warn("Failed to parse options for question:", question.title, e)
   434→          optionsString = question.options?.toString() || ""
   435→        }
   436→
   437→        return [
   438→          question.title,
   439→          question.content,
   440→          question.type,
   441→          optionsString,
   442→          question.correctAnswer,
   443→          question.explanation || "",
   444→          question.difficulty,
   445→          question.points.toString()
   446→        ]
   447→      })
   448→    ].map(row =>
   449→      // Properly escape CSV values that contain commas or quotes
   450→      row.map(cell => {
   451→        if (cell === null || cell === undefined) return ""
   452→        const str = cell.toString()
   453→        if (str.includes(",") || str.includes('"') || str.includes("\n")) {
   454→          return `"${str.replace(/"/g, '""')}"`
   455→        }
   456→        return str
   457→      }).join(",")
   458→    ).join("\n")
   459→
   460→    const blob = new Blob([csvContent], { type: "text/csv;charset=utf-8" })
   461→    const url = window.URL.createObjectURL(blob)
   462→    const a = document.createElement("a")
   463→    a.href = url
   464→    a.download = `${quizTitle.replace(/[^a-z0-9]/gi, '_').toLowerCase()}_questions.csv`
   465→    a.click()
   466→    window.URL.revokeObjectURL(url)
   467→    toast.success("Questions exported to CSV")
   468→  }
   469→
   470→  const handleImportQuestions = () => {

   472→  }
   473→
   474→  // Drag and drop handlers
   475→  const handleDragOver = (e: React.DragEvent) => {
   476→    e.preventDefault()
   477→    setIsDragOver(true)
   478→  }
   479→
   480→  const handleDragLeave = (e: React.DragEvent) => {

   484→
   485→  const handleDrop = (e: React.DragEvent) => {

   487→    setIsDragOver(false)

   489→    const files = e.dataTransfer.files

   513→
   514→  const handleImportWithGroup = async () => {
   515→    if (!importFile) {
   516→      toast.error("Please select a file to import")
   517→      return
   518→    }
   519→    
   520→    if (!selectedQuestionGroup) {
   521→      toast.error("Please select a question group")
   522→      return
   523→    }
   524→
   525→    setIsImporting(true)
   526→
   527→    Papa.parse(importFile, {
   528→      header: true,
   529→      skipEmptyLines: true,
   530→      complete: async (results) => {
   531→        try {
   532→
   533→          // Filter out empty rows and validate required fields
   534→          const validQuestions = results.data.filter((row: any) => {
   535→            const hasTitle = row.Title && row.Title.trim() !== ""
   536→            const hasContent = row.Content && row.Content.trim() !== ""
   537→            const hasType = row.Type && row.Type.trim() !== ""
   538→            const hasOptions = row.Options && row.Options.trim() !== ""
   539→            const hasCorrectAnswer = row["Correct Answer"] && row["Correct Answer"].trim() !== ""
   540→
   541→            return hasTitle && hasContent && hasType && hasOptions && hasCorrectAnswer
   542→          })
   543→
   544→
   545→          if (validQuestions.length === 0) {
   546→            toast.error("No valid questions found in CSV file. Please ensure all required fields are filled.")
   547→            return
   548→          }
   549→
   550→          // Create questions in the selected group
   551→          const importPromises = validQuestions.map(async (question: any, index: number) => {
   552→            try {
   553→
   554→              // Normalize question type
   555→              let questionType = question.Type?.toString().toUpperCase().trim()
   556→              if (!Object.values(QuestionType).includes(questionType as QuestionType)) {
   557→                questionType = QuestionType.MULTIPLE_CHOICE
   558→                console.warn(`Invalid question type "${question.Type}", defaulting to MULTIPLE_CHOICE`)
   559→              }
   560→
   561→              // Parse options
   562→              let options = []
   563→              if (question.Options) {
   564→                const optionsStr = question.Options.toString().trim()
   565→                if (optionsStr.startsWith('[') && optionsStr.endsWith(']')) {
   566→                  try {
   567→                    const parsed = JSON.parse(optionsStr)
   568→                    if (Array.isArray(parsed)) {
   569→                      options = parsed
   570→                    }
   571→                  } catch {
   572→                    options = optionsStr.split('|').map(opt => opt.trim())
   573→                  }
   574→                } else {
   575→                  options = optionsStr.split('|').map(opt => opt.trim())
   576→                }
   577→              }
   578→
   579→              // Create the question
   580→              const response = await fetch("/api/admin/question-groups", {
   581→                method: "POST",
   582→                headers: {
   583→                  "Content-Type": "application/json",
   584→                },
   585→                body: JSON.stringify({
   586→                  title: question.Title?.trim(),
   587→                  content: question.Content?.trim(),
   588→                  type: questionType,
   589→                  options: options,
   590→                  correctAnswer: question["Correct Answer"]?.toString(),
   591→                  explanation: question.Explanation?.trim() || "",
   592→                  difficulty: question.Difficulty?.toUpperCase() || "MEDIUM",
   593→                  groupId: selectedQuestionGroup
   594→                }),
   595→              })
   596→
   597→              if (!response.ok) {
   598→                const error = await response.json()
   599→                throw new Error(error.message || "Failed to create question")
   600→              }
   601→
   602→              return await response.json()
   603→            } catch (error) {
   604→              console.error(`Failed to import question ${index + 1}:`, error)
   605→              throw error
   606→            }
   607→          })
   608→
   609→          const importedQuestions = await Promise.all(importPromises)
   610→          
   611→          // Add imported questions to the quiz
   612→          const questionIds = importedQuestions.map(q => q.id)
   613→          const addResponse = await fetch(`/api/admin/quiz/${quizId}/questions`, {
   614→            method: "POST",
   615→            headers: {
   616→              "Content-Type": "application/json",
   617→            },
   618→            body: JSON.stringify({ questionIds }),
   619→          })
   620→
   621→          if (addResponse.ok) {
   622→            toast.success(`Successfully imported ${importedQuestions.length} questions and added to quiz`)
   623→            setIsImportSheetOpen(false)
   624→            setImportFile(null)
   625→            setSelectedQuestionGroup("")
   626→            fetchQuestions()
   627→            fetchAvailableQuestions()
   628→            setIsImporting(false)
   629→          } else {
   630→            toast.error("Questions were imported but failed to enroll to quiz")
   631→            setIsImporting(false)
   632→          }
   633→
   634→        } catch (error) {
   635→          console.error("Import error:", error)
   636→          toast.error(`Failed to import questions: ${error instanceof Error ? error.message : "Unknown error"}`)
   637→          setIsImporting(false)
   638→        }
   639→      },
   640→      error: (error) => {
   641→        console.error("CSV parsing error:", error)
   642→        toast.error("Failed to parse CSV file")
   643→        setIsImporting(false)
   644→      }
   645→    })
   646→  }
   647→
   648→  const handleFileChange = (event: React.ChangeEvent<HTMLInputElement>) => {
   649→    const file = event.target.files?.[0]
   650→    if (!file) return
   651→
   652→    Papa.parse(file, {
   653→      header: true,
   654→      skipEmptyLines: true,
   655→      complete: async (results) => {
