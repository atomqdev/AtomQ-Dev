    70→// Debounce hook to prevent too many API calls
    71→function useDebounce<T>(value: T, delay: number): T {
    72→  const [debouncedValue, setDebouncedValue] = useState<T>(value)
    73→
    74→  useEffect(() => {
    75→    const handler = setTimeout(() => {
    76→      setDebouncedValue(value)
    77→    }, delay)
    78→
    79→    return () => {
    80→      clearTimeout(handler)
    81→    }
    82→  }, [value, delay])
    83→
    84→  return debouncedValue
    85→}
