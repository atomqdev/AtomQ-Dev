   192→        method: 'DELETE'
   193→      })

   195→      if (response.ok) {
   196→        const data = await response.json()

   211→      toasts.actionFailed("Quiz data deletion")
   212→      setDeletionStatus(prev => ({ ...prev, data: 'pending' }))
   213→    }
   214→  }
   215→

   217→    if (!quizToDelete) return
   218→
   219→    const hasAttempts = (deleteInfo?.counts.attempts || 0) > 0
   220→    const hasTabSwitches = (deleteInfo?.counts.tabSwitches || 0) > 0
   221→    if ((hasAttempts || hasTabSwitches) && deletionStatus.data !== 'deleted') {
   222→      toasts.error('Please delete quiz data first')
   223→      return
   224→    }
   225→
   226→    try {
   227→      setDeletionStatus(prev => ({ ...prev, questions: 'deleting' }))
   228→      const response = await fetch(`/api/admin/quiz/${quizToDelete.id}/unenroll-questions`, {
   229→        method: 'DELETE'
   230→      })
   231→
   232→      if (response.ok) {
   233→        const data = await response.json()
   234→        toasts.success(`${data.count.questions || 0} question(s) unenrolled successfully`)
   235→        setDeletionStatus(prev => ({ ...prev, questions: 'deleted' }))
   236→
   237→        const refreshResponse = await fetch(`/api/admin/quiz/${quizToDelete.id}/delete-info`)
   238→        if (refreshResponse.ok) {
   239→          const refreshData = await refreshResponse.json()
   240→          setDeleteInfo(refreshData)
   241→        }
   242→      } else {
   243→        toasts.actionFailed("Question unenrollment")
   244→        setDeletionStatus(prev => ({ ...prev, questions: 'pending' }))
   245→      }
   246→    } catch (error) {
   247→      console.error("Error unenrolling questions:", error)
   248→      toasts.actionFailed("Question unenrollment")

   255→

   257→      toasts.error('Please unenroll questions first')
   258→      return
   259→    }
   260→
   261→    try {
   262→      setDeletionStatus(prev => ({ ...prev, users: 'deleting' }))
   263→      const response = await fetch(`/api/admin/quiz/${quizToDelete.id}/unenroll-users`, {
   264→        method: 'DELETE'
   265→      })
   266→
   267→      if (response.ok) {
   268→        const data = await response.json()
   269→        toasts.success(`${data.count.users || 0} user(s) unenrolled successfully`)
   270→        setDeletionStatus(prev => ({ ...prev, users: 'deleted' }))
   271→
   272→        const refreshResponse = await fetch(`/api/admin/quiz/${quizToDelete.id}/delete-info`)
   273→        if (refreshResponse.ok) {
   274→          const refreshData = await refreshResponse.json()
   275→          setDeleteInfo(refreshData)
   276→        }
   277→      } else {
   278→        toasts.actionFailed("User unenrollment")
   279→        setDeletionStatus(prev => ({ ...prev, users: 'pending' }))
   280→      }
   281→    } catch (error) {
   282→      console.error("Error unenrolling users:", error)
   283→      toasts.actionFailed("User unenrollment")
   284→      setDeletionStatus(prev => ({ ...prev, users: 'pending' }))
   285→    }
   286→  }
   287→
   288→  const handleDeleteQuiz = async (quizId: string) => {
   289→    if (!quizToDelete || deleteConfirmation !== "CONFIRM DELETE") {
   290→      toasts.error('Please type "CONFIRM DELETE" to confirm deletion')
   291→      return
   292→    }
   293→
   294→    const hasAttempts = (deleteInfo?.counts.attempts || 0) > 0
   295→    const hasTabSwitches = (deleteInfo?.counts.tabSwitches || 0) > 0
   296→    const hasQuestions = (deleteInfo?.counts.questions || 0) > 0
   297→    const hasUsers = (deleteInfo?.counts.users || 0) > 0
   298→
   299→    if ((hasAttempts || hasTabSwitches) && deletionStatus.data !== 'deleted') {
   300→      toasts.error('Please delete quiz data first')
   301→      return
   302→    }
   303→
   304→    if (hasQuestions && deletionStatus.questions !== 'deleted') {
   305→      toasts.error('Please unenroll questions first')
   306→      return
   307→    }
   308→
   309→    if (hasUsers && deletionStatus.users !== 'deleted') {
   310→      toasts.error('Please unenroll users first')
   311→      return
   312→    }
   313→
   314→    try {
   315→      setDeleteLoading(quizId)
   316→      const response = await fetch(`/api/admin/quiz/${quizId}`, {
   317→        method: "DELETE",
   318→      })
   319→
   320→      if (response.ok) {
   321→        toasts.success(`${quizToDelete?.title || "Quiz"} deleted successfully`)
   322→        setQuizzes(quizzes.filter(q => q.id !== quizId))
   323→        setIsDeleteDialogOpen(false)
   324→        setQuizToDelete(null)
   325→        setDeleteInfo(null)
   326→        setDeleteConfirmation("")
   327→        setDeletionStatus({ data: 'pending', questions: 'pending', users: 'pending' })
   328→        fetchGroup()
   329→      } else {
   330→        const error = await response.json()
   331→        toasts.error(error.message || "Quiz deletion failed")
   332→      }
   333→    } catch (error) {
   334→      console.error("Error deleting quiz:", error)
   335→      toasts.actionFailed("Quiz deletion")
   336→    } finally {
   337→      setDeleteLoading(null)
   338→    }
   339→  }
   340→
   341→  const columns: ColumnDef<Quiz>[] = [

   350→            Title

   352→          </Button>
   353→        )
   354→      },
   355→      cell: ({ row }) => (
   356→        <div className="font-medium">{row.getValue("title")}</div>
   357→      ),
   358→    },
   359→    {
   360→      accessorKey: "difficulty",
   361→      header: "Difficulty",

   373→    {

   375→      header: "Status",
   376→      cell: ({ row }) => {
   377→        const status = row.getValue("status") as QuizStatus
   378→        return (

   380→            {status}
   381→          </Badge>

   383→      },
   384→    },
   385→    {
   386→      accessorKey: "timeLimit",
   387→      header: "Time Limit",
   388→      cell: ({ row }) => {
   389→        const timeLimit = row.getValue("timeLimit") as number | null
   390→        return timeLimit ? `${timeLimit} min` : "-"

   393→    {

   395→      header: "Questions",
   396→      cell: ({ row }) => {
   397→        const quiz = row.original
   398→        return quiz._count?.quizQuestions || 0
   399→      },
   400→    },
   401→    {
   402→      accessorKey: "_count.quizAttempts",
   403→      header: "Attempts",
   404→      cell: ({ row }) => {
   405→        const quiz = row.original
   406→        return quiz._count?.quizAttempts || 0

   410→      accessorKey: "createdAt",
   411→      header: ({ column }) => {

   415→            onClick={() => column.toggleSorting(column.getIsSorted() === "asc")}

   448→      },

   450→    {
   451→      id: "manageUsers",

   454→      cell: ({ row }) => {
   455→        const quiz = row.original
   456→        return (
   457→          <Tooltip>
   458→            <TooltipTrigger asChild>
   459→              <Button
   460→                variant="ghost"
   461→                size="icon"
   462→                className="h-8 w-8"
   463→                onClick={() => router.push(`/admin/quiz/${quiz.id}/users`)}
   464→              >
   465→                <Users className="h-4 w-4" />

   480→          <Tooltip>
   481→            <TooltipTrigger asChild>
   482→              <Button
   483→                variant="ghost"

   525→
   526→  const fetchGroup = async () => {
   527→    try {
   528→      const response = await fetch(`/api/admin/quiz-groups/${id}`)
   529→      if (response.ok) {
   530→        const data = await response.json()
   531→        setGroup(data)
   532→      } else if (response.status === 404) {
   533→        toasts.error("Quiz group not found")
   534→        router.push("/admin/quiz-group")
   535→      }
   536→    } catch (error) {
   537→      toasts.networkError()
   538→    }
   539→  }
   540→
   541→  const fetchQuizzes = async () => {
   542→    try {
   543→      const response = await fetch(
   544→        `/api/admin/quiz?groupId=${id}&page=1&pageSize=100`
   545→      )
   546→      if (response.ok) {
   547→        const data = await response.json()
   548→        setQuizzes(data.quizzes || [])
   549→      }
   550→    } catch (error) {
   551→      toasts.networkError()
   552→    } finally {
   553→      setLoading(false)
   554→    }
   555→  }
   556→
   557→  const handleCreateQuiz = async (e: React.FormEvent) => {
   558→    e.preventDefault()
   559→    if (!formData.title.trim()) {
   560→      toasts.error("Title is required")
   561→      return
   562→    }
   563→    if (
   564→      formData.startDate &&
   565→      formData.endDate &&
   566→      new Date(formData.endDate) <= new Date(formData.startDate)
   567→    ) {
   568→      toasts.error("End date must be after start date")
   569→      return
   570→    }
   571→    setSubmitLoading(true)
   572→
   573→    try {
   574→      const response = await fetch("/api/admin/quiz", {
   575→        method: "POST",
   576→        headers: {
   577→          "Content-Type": "application/json",
   578→        },
   579→        body: JSON.stringify({
   580→          ...formData,
   581→          timeLimit: formData.timeLimit ? parseInt(formData.timeLimit) : null,
   582→          negativePoints: formData.negativePoints
   583→            ? parseFloat(formData.negativePoints)
   584→            : null,
   585→          maxAttempts:
   586→            formData.maxAttempts === "" ? null : parseInt(formData.maxAttempts),
   587→          startDate: formatDateToUTC(formData.startDate),
   588→          endDate: formatDateToUTC(formData.endDate),
   589→          groupId: id,
   590→        }),
   591→      })
   592→
   593→      if (response.ok) {
   594→        toasts.success("Quiz created successfully in this group")
   595→        setIsAddDialogOpen(false)
   596→        setFormData({
   597→          title: "",
   598→          description: "",
   599→          timeLimit: "",
   600→          difficulty: DifficultyLevel.EASY,
   601→          status: QuizStatus.DRAFT,
   602→          negativeMarking: false,
   603→          negativePoints: "",
   604→          randomOrder: false,
   605→          maxAttempts: "",
   606→          startDate: "",
   607→          endDate: "",
   608→          checkAnswerEnabled: false,
   609→        })
   610→        fetchQuizzes()
   611→        fetchGroup()
   612→      } else {
   613→        const error = await response.json()
   614→        toasts.error(error.message || "Failed to create quiz")
   615→      }
   616→    } catch (error) {
   617→      toasts.networkError()
   618→    } finally {
   619→      setSubmitLoading(false)
   620→    }
   621→  }
   622→
   623→  if (loading) {
   624→    return (
   625→      <div className="flex items-center justify-center h-[80vh]">
   626→        <HexagonLoader size={80} />
   627→      </div>
   628→    )
   629→  }
   630→
   631→  return (
   632→    <div className="space-y-6">
   633→      <div className="flex items-center gap-4">
   634→        <Button
   635→          variant="ghost"
   636→          size="icon"
   637→          onClick={() => router.push("/admin/quiz-group")}
   638→        >
   639→          <ArrowLeft className="h-5 w-5" />
   640→        </Button>
   641→        <div className="flex-1">
   642→          <div className="flex items-center gap-3">
