    63→    63→const formatDateDDMMYYYY = (dateString: string) => {
    64→    64→  const date = new Date(dateString)
    65→    65→  const day = String(date.getDate()).padStart(2, '0')
    66→    66→  const month = String(date.getMonth() + 1).padStart(2, '0')
    67→    67→  const year = date.getFullYear()
    68→    68→  return `${day}/${month}/${year}`
