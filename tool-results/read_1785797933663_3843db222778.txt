    62→// Helper function to format dates in dd/mm/yyyy format
    63→const formatDateDDMMYYYY = (dateString: string) => {
    64→  const date = new Date(dateString)
    65→  const day = String(date.getDate()).padStart(2, '0')
    66→  const month = String(date.getMonth() + 1).padStart(2, '0')
    67→  const year = date.getFullYear()
    68→  return `${day}/${month}/${year}`
    69→}

   626→                {((deleteInfo.counts.quizQuestions || 0) > 0 || (deleteInfo.counts.assessmentQuestions || 0) > 0) && (

   639→                    </div>
   640→                    <div className="flex items-center gap-2">

   657→                              <Trash2 className="h-4 w-4" />
   658→                            )}
   659→                          </Button>
   660→                        </TooltipTrigger>
   661→                        <TooltipContent>
   662→                          {deletionStatus.unassign === 'deleted' ? 'Questions unassigned' : deletionStatus.unassign === 'deleting' ? 'Unassigning...' : 'Unassign questions'}

   664→                      </Tooltip>
   665→                    </div>
   666→                  </div>
   667→                )}
   668→
   669→                {/* Step 2: Delete Questions */}
   670→                {deleteInfo.counts.questions > 0 && (
   671→                  <div className="flex items-center justify-between p-4 border rounded-lg bg-muted/30">
   672→                    <div className="flex items-center gap-3">

   686→                    <div className="flex items-center gap-2">

   688→                        <CheckCircle className="w-5 h-5 text-green-600" />
   689→                      )}
   690→                      <Tooltip>
   691→                        <TooltipTrigger asChild>
   692→                          <Button
   693→                            onClick={handleDeleteQuestions}
   694→                            disabled={
   695→                              deletionStatus.questions === 'deleted' ||
   696→                              deletionStatus.questions === 'deleting' ||
   697→                              (((deleteInfo.counts.quizQuestions || 0) > 0 || (deleteInfo.counts.assessmentQuestions || 0) > 0) && deletionStatus.unassign !== 'deleted')
   698→                            }
   699→                            variant={deletionStatus.questions === 'deleted' ? 'outline' : 'destructive'}
   700→                            size="icon"
   701→                          >
   702→                            {deletionStatus.questions === 'deleting' ? (
   703→                              <Loader2 className="h-4 w-4 animate-spin" />
   704→                            ) : deletionStatus.questions === 'deleted' ? (
   705→                              <CheckCircle className="h-4 w-4" />
   706→                            ) : (
   707→                              <Trash2 className="h-4 w-4" />
   708→                            )}
   709→                          </Button>
   710→                        </TooltipTrigger>
   711→                        <TooltipContent>
   712→                          {deletionStatus.questions === 'deleted' ? 'Questions deleted' : deletionStatus.questions === 'deleting' ? 'Deleting...' : 'Delete questions'}

   714→                      </Tooltip>
   715→                    </div>
   716→                  </div>
   717→                )}
   718→
   719→                {/* Empty State - Ready to delete */}
   720→                {deleteInfo.counts.questions === 0 &&
   721→                 (deleteInfo.counts.quizQuestions || 0) === 0 &&
   722→                 (deleteInfo.counts.assessmentQuestions || 0) === 0 && (
   723→                  <div className="p-4 border rounded-lg bg-green-50 border-green-200">
   724→                    <p className="text-green-800 text-sm font-medium">
   725→                      ✓ All critical data removed. Ready to delete question group.
   726→                    </p>
   727→                  </div>
   728→                )}
   729→              </div>
   730→            ) : (
   731→              <div className="flex items-center justify-center p-8">
   732→                <Loader2 className="h-8 w-8 animate-spin text-muted-foreground" />
   733→              </div>
   734→            )}
   735→          </div>
   736→          </TooltipProvider>
   737→
   738→          {/* Final Confirmation Input */}
   739→          <div className="mt-4 pt-4 border-t space-y-2">
   740→            <Label htmlFor="delete-confirmation">
   741→              <span className="font-semibold text-destructive">CONFIRM DELETE</span> to proceed:
   742→            </Label>
   743→            <Input
   744→              id="delete-confirmation"
   745→              value={deleteConfirmation}
   746→              onChange={(e) => setDeleteConfirmation(e.target.value)}
   747→              placeholder="CONFIRM DELETE"
   748→              autoComplete="off"
   749→              className="uppercase"
   750→            />
   751→          </div>
   752→
   753→          <AlertDialogFooter>
   754→            <AlertDialogCancel onClick={() => {
   755→              setIsDeleteDialogOpen(false)
   756→              setGroupToDelete(null)
   757→              setDeleteInfo(null)
   758→              setDeleteConfirmation("")
   759→              setDeletionStatus({ unassign: 'pending', questions: 'pending' })
   760→            }}>
   761→              Cancel
   762→            </AlertDialogCancel>
   763→            <AlertDialogAction
   764→              onClick={() => groupToDelete && handleDeleteGroup(groupToDelete.id)}
   765→              className="bg-red-600 hover:bg-red-700"
   766→              disabled={
   767→                deleteLoading !== null ||
   768→                deleteConfirmation !== "CONFIRM DELETE"
   769→              }
   770→            >
   771→              {deleteLoading === groupToDelete?.id ? (
   772→                <>
   773→                  <Loader2 className="mr-2 h-4 w-4 animate-spin" />
   774→                  Deleting...
   775→                </>
   776→              ) : (
   777→                "Delete Question Group"
   778→              )}
   779→            </AlertDialogAction>
   780→          </AlertDialogFooter>
   781→        </AlertDialogContent>
   782→      </AlertDialog>
   783→    </div>
   784→  )
