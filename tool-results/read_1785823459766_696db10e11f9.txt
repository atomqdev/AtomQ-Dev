   296→    }

   302→        await document.exitFullscreen()

   304→        lastFullscreenStateRef.current = false

   362→          setSwitchesRemaining(data.switchesRemaining)
   363→        }
   364→
   365→        // Check if we should auto-submit (regardless of response status)
   366→        if (data.shouldAutoSubmit) {
   367→          triggerAutoSubmit()
   368→        } else if (response.ok) {
   369→          // Only show warnings if the request was successful
   370→          if (data.switchesRemaining !== null && data.switchesRemaining <= 1) {

   376→      } catch (error) {
   377→        console.error("Error recording fullscreen exit:", error)
   378→      }
   379→      tabSwitchDebounceRef.current = null

   546→            // Submit immediately without any warning or delay
   547→            handleSubmit(true) // Time expiry should also be marked as auto-submitted
   548→          }
   549→          return newTime
   550→        })
   551→      }, 1000)
   552→    }

   554→    return () => {

   556→        clearInterval(intervalRef.current)
   557→        intervalRef.current = null

   585→      }

   590→  }, [handleFullscreenChange, handleVisibilityChange])
   591→
   592→  // ==================== SECURITY FEATURES ====================
   593→
   594→  const handleContextMenu = useCallback((e: Event) => {
   595→    // Always disable right-click on assessment take page

   599→  const handleKeyDown = useCallback((e: KeyboardEvent) => {
   600→    // Always block these shortcuts on assessment take page regardless of disableCopyPaste setting
   601→
   602→    // Block special keys
   603→    const specialKeys = ['F12', 'Insert', 'PrintScreen', 'ScrollLock', 'Pause']
   604→    const key = e.key
   605→
   606→    if (specialKeys.includes(key) || key === 'F12') {
   607→      e.preventDefault()
   608→      toasts.warning("This action is not allowed during the assessment")
   609→      return
   610→    }
   611→
   612→    // Get modifier keys
   613→    const ctrlKey = e.ctrlKey
   614→    const metaKey = e.metaKey // macOS Command key
   615→    const shiftKey = e.shiftKey
   616→    const altKey = e.altKey // Alt key (Windows) / Option key (macOS)
   617→
   618→    // Check if any modifier is pressed
   619→    const hasModifier = ctrlKey || metaKey || shiftKey || altKey
   620→
   621→    // Build key combo string (Alt on Windows = Option on macOS)

   762→      const heightDiff = window.outerHeight - window.innerHeight
   763→

   765→      // In incognito mode, small differences can occur, but DevTools creates larger ones

   767→        devToolsDetected = true
   768→      }
