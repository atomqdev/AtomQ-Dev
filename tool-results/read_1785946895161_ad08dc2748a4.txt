    56→const formatDateDDMMYYYY = (dateString: string) => {
    57→  const date = new Date(dateString)
    58→  const day = String(date.getDate()).padStart(2, '0')
    59→  const month = String(date.getMonth() + 1).padStart(2, '0')
    60→  const year = date.getFullYear()
    61→  return `${day}/${month}/${year}`
    62→}

   602→  const handleJsonImport = async (event: React.ChangeEvent<HTMLInputElement>) => {
   603→    const file = event.target.files?.[0]
   604→    if (!file) return

   606→    // Reset the input so the same file can be selected again
   607→    event.target.value = ""

   609→    if (!file.name.endsWith('.json')) {
   610→      toast.error("Please select a JSON file")
   611→      return
   612→    }
   613→
   614→    setJsonImportLoading(true)
   615→    try {
   616→      const text = await file.text()
   617→      let parsedData: unknown
   618→      try {
   619→        parsedData = JSON.parse(text)
   620→      } catch {
   621→        toast.error("Invalid JSON file. Please check the file format.")
   622→        return
   623→      }
   624→
   625→      // Support both a raw array and an object with an `importData`/`questions` array
   626→      let importArray: any[] = []
   627→      if (Array.isArray(parsedData)) {
