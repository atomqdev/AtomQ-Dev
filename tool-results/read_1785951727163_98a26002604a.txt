   137→function SortableQuestion({
   138→  question,
   139→  onEdit,
   140→  onDelete,
   141→  onView
   142→}: {
   143→  question: Question
   144→  onEdit: (question: Question) => void
   145→  onDelete: (questionId: string) => void
   146→  onView: (question: Question) => void
   147→}) {
   148→  const {
   149→    attributes,
   150→    listeners,
   151→    setNodeRef,
   152→    transform,
   153→    transition,
   154→    isDragging,
   155→  } = useSortable({ id: question.id })
   156→
   157→  const style = {
   158→    transform: CSS.Transform.toString(transform),
   159→    transition,
   160→    opacity: isDragging ? 0.5 : 1,
   161→  }
   162→
   163→  return (
   164→    <TableRow ref={setNodeRef} style={style} {...attributes}>
   165→      <TableCell>
   166→        <div {...listeners} className="cursor-grab">
   167→          <GripVertical className="h-4 w-4" />
   168→        </div>
   169→      </TableCell>
   170→      <TableCell className="font-medium">{question.order}</TableCell>
   171→      <TableCell>
   172→        <div>
   173→          <div className="font-medium">{question.reference}</div>
   174→          {/* <div className="text-sm text-muted-foreground">
   175→            <RichTextDisplay content={question.title} />
   176→          </div> */}
   177→        </div>
   178→      </TableCell>
   179→      <TableCell>
   180→        <Badge variant={
   181→          question.type === QuestionType.MULTIPLE_CHOICE ? "default" :
   182→            question.type === QuestionType.MULTI_SELECT ? "secondary" :
   183→            question.type === QuestionType.TRUE_FALSE ? "outline" : "destructive"
   184→        }>
   185→          {question.type.replace('_', ' ')}
   186→          {question.type === QuestionType.MULTI_SELECT && (() => {
   187→            const sc = getMultiSelectCount(question.correctAnswer || '')
   188→            return sc > 0 ? ` (Select ${sc})` : ''
   189→          })()}
   190→        </Badge>
   191→      </TableCell>
   192→      <TableCell>
   193→        <Badge variant={
   194→          question.difficulty === DifficultyLevel.EASY ? "default" :
   195→            question.difficulty === DifficultyLevel.MEDIUM ? "secondary" : "destructive"
   196→        }>
   197→          {question.difficulty}
   198→        </Badge>
   199→      </TableCell>
   200→      <TableCell>{question.points}</TableCell>
   201→      <TableCell>
   202→        <div className="flex gap-2">
   203→          <Button
   204→            variant="ghost"
   205→            size="sm"
   206→            onClick={() => onView(question)}
   207→          >
   208→            <Eye className="h-4 w-4" />
   209→          </Button>
   210→
   211→          <AlertDialog>
   212→            <AlertDialogTrigger asChild>
   213→              <Button
   214→                variant="ghost"
   215→                size="sm"
   216→                className="text-red-600"
   217→              >
   218→                <Trash2 className="h-4 w-4" />
   219→              </Button>
   220→            </AlertDialogTrigger>
   221→            <AlertDialogContent>
   222→              <AlertDialogHeader>
   223→                <AlertDialogTitle>Remove Question</AlertDialogTitle>
   224→                <AlertDialogDescription>
   225→                  Are you sure you want to remove this question from the quiz? This action cannot be undone.
   226→                </AlertDialogDescription>
   227→              </AlertDialogHeader>
   228→              <AlertDialogFooter>
   229→                <AlertDialogCancel>Cancel</AlertDialogCancel>
   230→                <AlertDialogAction onClick={() => onDelete(question.id)}>
   231→                  Remove
   232→                </AlertDialogAction>
   233→              </AlertDialogFooter>
   234→            </AlertDialogContent>
   235→          </AlertDialog>
   236→        </div>
   237→      </TableCell>
   238→    </TableRow>
   239→  )
   240→}

   426→  const handleExportQuestions = () => {

   428→      ["Reference", "Title", "Type", "Options", "Correct Answer", "Explanation", "Difficulty", "Points"],
   429→      ...questions.map(question => {

   433→          const parsedOptions = Array.isArray(question.options)
   434→            ? question.options

   439→          optionsString = question.options?.toString() || ""

   441→
   442→        return [
   443→          question.reference,
   444→          question.title,
   445→          question.type,
   446→          optionsString,
   447→          question.correctAnswer,
   448→          question.explanation || "",
   449→          question.difficulty,

   533→      header: true,
   534→      skipEmptyLines: true,
   535→      complete: async (results) => {
   536→        try {
   537→
   538→          // Filter out empty rows and validate required fields
   539→          const validQuestions = results.data.filter((row: any) => {
   540→            const hasReference = row.Reference && row.Reference.trim() !== ""
   541→            const hasTitle = row.Title && row.Title.trim() !== ""
   542→            const hasType = row.Type && row.Type.trim() !== ""
   543→            const hasOptions = row.Options && row.Options.trim() !== ""
   544→            const hasCorrectAnswer = row["Correct Answer"] && row["Correct Answer"].trim() !== ""
   545→
   546→            return hasReference && hasTitle && hasType && hasOptions && hasCorrectAnswer
   547→          })
   548→
   549→

   551→            toast.error("No valid questions found in CSV file. Please ensure all required fields are filled.")
   552→            return
   553→          }
   554→
   555→          // Create questions in the selected group
   556→          const importPromises = validQuestions.map(async (question: any, index: number) => {
   557→            try {
   558→
   559→              // Normalize question type
   560→              let questionType = question.Type?.toString().toUpperCase().trim()
   561→              if (!Object.values(QuestionType).includes(questionType as QuestionType)) {
   562→                questionType = QuestionType.MULTIPLE_CHOICE
   563→                console.warn(`Invalid question type "${question.Type}", defaulting to MULTIPLE_CHOICE`)
   564→              }
   565→
   566→              // Parse options
   567→              let options = []
   568→              if (question.Options) {
   569→                const optionsStr = question.Options.toString().trim()
   570→                if (optionsStr.startsWith('[') && optionsStr.endsWith(']')) {
   571→                  try {
   572→                    const parsed = JSON.parse(optionsStr)
   573→                    if (Array.isArray(parsed)) {
   574→                      options = parsed
   575→                    }
   576→                  } catch {
   577→                    options = optionsStr.split('|').map(opt => opt.trim())
   578→                  }
   579→                } else {
   580→                  options = optionsStr.split('|').map(opt => opt.trim())
   581→                }
   582→              }
   583→
   584→              // Create the question
   585→              const response = await fetch("/api/admin/question-groups", {
   586→                method: "POST",
   587→                headers: {
   588→                  "Content-Type": "application/json",
   589→                },
   590→                body: JSON.stringify({
   591→                  reference: question.Reference?.trim(),
   592→                  title: question.Title?.trim(),
   593→                  type: questionType,
   594→                  options: options,
   595→                  correctAnswer: question["Correct Answer"]?.toString(),
   596→                  explanation: question.Explanation?.trim() || "",
   597→                  difficulty: question.Difficulty?.toUpperCase() || "MEDIUM",
   598→                  groupId: selectedQuestionGroup
   599→                }),
   600→              })
   601→
   602→              if (!response.ok) {
   603→                const error = await response.json()
   604→                throw new Error(error.message || "Failed to create question")
   605→              }
   606→
   607→              return await response.json()
   608→            } catch (error) {
   609→              console.error(`Failed to import question ${index + 1}:`, error)
   610→              throw error
   611→            }
   612→          })
   613→
   614→          const importedQuestions = await Promise.all(importPromises)
   615→          
   616→          // Add imported questions to the quiz
   617→          const questionIds = importedQuestions.map(q => q.id)
   618→          const addResponse = await fetch(`/api/admin/quiz/${quizId}/questions`, {
   619→            method: "POST",
   620→            headers: {
   621→              "Content-Type": "application/json",
   622→            },
   623→            body: JSON.stringify({ questionIds }),
   624→          })
   625→
   626→          if (addResponse.ok) {

   628→            setIsImportSheetOpen(false)
   629→            setImportFile(null)
   630→            setSelectedQuestionGroup("")

   636→            setIsImporting(false)
   637→          }
   638→
   639→        } catch (error) {
   640→          console.error("Import error:", error)
   641→          toast.error(`Failed to import questions: ${error instanceof Error ? error.message : "Unknown error"}`)
   642→          setIsImporting(false)
   643→        }
   644→      },
   645→      error: (error) => {
   646→        console.error("CSV parsing error:", error)
   647→        toast.error("Failed to parse CSV file")
   648→        setIsImporting(false)
   649→      }
   650→    })
   651→  }
