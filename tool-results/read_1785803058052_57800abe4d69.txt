    61→// Helper function to format dates in dd/mm/yyyy format
    62→const formatDateDDMMYYYY = (dateString: string) => {
    63→  const date = new Date(dateString)
    64→  const day = String(date.getDate()).padStart(2, '0')
    65→  const month = String(date.getMonth() + 1).padStart(2, '0')
    66→  const year = date.getFullYear()
    67→  return `${day}/${month}/${year}`
    68→}

   615→                            onClick={handleUnassignQuestions}
   616→                            disabled={deletionStatus.unassign === 'deleted' || deletionStatus.unassign === 'deleting'}
   617→                            variant={deletionStatus.unassign === 'deleted' ? 'outline' : 'destructive'}
   618→                            size="icon"
   619→                          >
   620→                            {deletionStatus.unassign === 'deleting' ? (
   621→                              <Loader2 className="h-4 w-4 animate-spin" />
   622→                            ) : deletionStatus.unassign === 'deleted' ? (
   623→                              <CheckCircle className="h-4 w-4" />
   624→                            ) : (
   625→                              <Trash2 className="h-4 w-4" />
   626→                            )}
   627→                          </Button>
   628→                        </TooltipTrigger>
   629→                        <TooltipContent>
   630→                          {deletionStatus.unassign === 'deleted' ? 'Questions unassigned' : deletionStatus.unassign === 'deleting' ? 'Unassigning...' : 'Unassign questions'}
   631→                        </TooltipContent>
   632→                      </Tooltip>
   633→                    </div>
   634→                  </div>
   635→                )}
   636→
   637→                {/* Step 2: Delete Questions */}
   638→                {deleteInfo.counts.questions > 0 && (
   639→                  <div className="flex items-center justify-between p-4 border rounded-lg bg-muted/30">
   640→                    <div className="flex items-center gap-3">
   641→                      <BookOpen className="w-5 h-5 text-blue-600" />
   642→                      <div>
   643→                        <p className="font-medium">Questions</p>
   644→                        <p className="text-sm text-muted-foreground">
   645→                          {deleteInfo.counts.questions} question{deleteInfo.counts.questions !== 1 ? 's' : ''} in this group
   646→                        </p>
   647→                        {deleteInfo.counts.reportedQuestions > 0 && (
   648→                          <p className="text-xs text-yellow-600">
   649→                            Includes {deleteInfo.counts.reportedQuestions} reported question(s)
   650→                          </p>
   651→                        )}
   652→                      </div>
   653→                    </div>
   654→                    <div className="flex items-center gap-2">
   655→                      {deletionStatus.questions === 'deleted' && (
   656→                        <CheckCircle className="w-5 h-5 text-green-600" />
   657→                      )}
   658→                      <Tooltip>
   659→                        <TooltipTrigger asChild>
   660→                          <Button
   661→                            onClick={handleDeleteQuestions}
   662→                            disabled={
   663→                              deletionStatus.questions === 'deleted' ||
   664→                              deletionStatus.questions === 'deleting' ||
   665→                              (((deleteInfo.counts.quizQuestions || 0) > 0 || (deleteInfo.counts.assessmentQuestions || 0) > 0) && deletionStatus.unassign !== 'deleted')
   666→                            }
   667→                            variant={deletionStatus.questions === 'deleted' ? 'outline' : 'destructive'}
   668→                            size="icon"
   669→                          >
   670→                            {deletionStatus.questions === 'deleting' ? (
   671→                              <Loader2 className="h-4 w-4 animate-spin" />
   672→                            ) : deletionStatus.questions === 'deleted' ? (
   673→                              <CheckCircle className="h-4 w-4" />
   674→                            ) : (
   675→                              <Trash2 className="h-4 w-4" />
   676→                            )}
   677→                          </Button>
   678→                        </TooltipTrigger>
   679→                        <TooltipContent>
   680→                          {deletionStatus.questions === 'deleted' ? 'Questions deleted' : deletionStatus.questions === 'deleting' ? 'Deleting...' : 'Delete questions'}
   681→                        </TooltipContent>
   682→                      </Tooltip>
   683→                    </div>
   684→                  </div>
   685→                )}
   686→
   687→                {/* Empty State - Ready to delete */}
   688→                {deleteInfo.counts.questions === 0 &&
   689→                 (deleteInfo.counts.quizQuestions || 0) === 0 &&
   690→                 (deleteInfo.counts.assessmentQuestions || 0) === 0 && (
   691→                  <div className="p-4 border rounded-lg bg-green-50 border-green-200">
   692→                    <p className="text-green-800 text-sm font-medium">
   693→                      ✓ All critical data removed. Ready to delete question group.
   694→                    </p>
   695→                  </div>
   696→                )}
   697→              </div>
   698→            ) : (
   699→              <div className="flex items-center justify-center p-8">
   700→                <Loader2 className="h-8 w-8 animate-spin text-muted-foreground" />
   701→              </div>
   702→            )}
   703→          </div>
   704→          </TooltipProvider>
   705→
   706→          {/* Final Confirmation Input */}
   707→          <div className="mt-4 pt-4 border-t space-y-2">
   708→            <Label htmlFor="delete-confirmation">
   709→              <span className="font-semibold text-destructive">CONFIRM DELETE</span> to proceed:
   710→            </Label>
   711→            <Input
   712→              id="delete-confirmation"
   713→              value={deleteConfirmation}
   714→              onChange={(e) => setDeleteConfirmation(e.target.value)}
   715→              placeholder="CONFIRM DELETE"
   716→              autoComplete="off"
   717→              className="uppercase"
   718→            />
   719→          </div>
   720→
   721→          <AlertDialogFooter>
   722→            <AlertDialogCancel onClick={() => {
   723→              setIsDeleteDialogOpen(false)
   724→              setGroupToDelete(null)
   725→              setDeleteInfo(null)
   726→              setDeleteConfirmation("")
   727→              setDeletionStatus({ unassign: 'pending', questions: 'pending' })
   728→            }}>
   729→              Cancel
   730→            </AlertDialogCancel>
   731→            <AlertDialogAction
   732→              onClick={() => groupToDelete && handleDeleteGroup(groupToDelete.id)}
   733→              className="bg-red-600 hover:bg-red-700"
   734→              disabled={
   735→                deleteLoading !== null ||
   736→                deleteConfirmation !== "CONFIRM DELETE"
   737→              }
   738→            >
   739→              {deleteLoading === groupToDelete?.id ? (
   740→                <>
   741→                  <Loader2 className="mr-2 h-4 w-4 animate-spin" />
   742→                  Deleting...
   743→                </>
   744→              ) : (
   745→                "Delete Question Group"
   746→              )}
   747→            </AlertDialogAction>
   748→          </AlertDialogFooter>
   749→        </AlertDialogContent>
   750→      </AlertDialog>
   751→    </div>
   752→  )
   753→}
