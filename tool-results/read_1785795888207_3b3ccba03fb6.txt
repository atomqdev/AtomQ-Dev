    68→// Debounce hook to prevent too many API calls
    69→function useDebounce<T>(value: T, delay: number): T {
    70→  const [debouncedValue, setDebouncedValue] = useState<T>(value)
    71→
    72→  useEffect(() => {
    73→    const handler = setTimeout(() => {
    74→      setDebouncedValue(value)
    75→    }, delay)
    76→
    77→    return () => {
    78→      clearTimeout(handler)
    79→    }
    80→  }, [value, delay])
    81→
    82→  return debouncedValue
    83→}
